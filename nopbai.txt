package com.example.monstersurvivor.android;

import android.content.Context;
import android.content.Intent;
import android.os.Bundle;
import android.view.Menu;
import android.view.MenuItem;
import android.view.View;
import android.view.animation.Animation;
import android.view.animation.AnimationUtils;
import android.widget.Toast;

import androidx.annotation.NonNull;
import androidx.appcompat.app.AlertDialog;
import androidx.appcompat.app.AppCompatActivity;

import com.example.monstersurvivor.R;
import com.example.monstersurvivor.android.model.GamePreferences;
import com.example.monstersurvivor.android.util.UiUtils;
import com.google.android.material.appbar.MaterialToolbar;
import com.google.android.material.bottomnavigation.BottomNavigationView;

/**
 * Main Home Activity for Monster Survivor.
 * Acts as the primary LAUNCHER entry point. Supports bilingual localization.
 */
public class HomeActivity extends NativeAudioActivity {

    private BottomNavigationView mBottomNav;
    private View mPanelTitle;
    private View mLayoutButtons;
    private String mCurrentLanguage;

    @Override
    protected void attachBaseContext(Context newBase) {
        super.attachBaseContext(UiUtils.wrapLocale(newBase));
    }

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        UiUtils.applyAppTheme(this);
        super.onCreate(savedInstanceState);
        UiUtils.applyImmersiveMode(this);
        setContentView(R.layout.activity_home);

        mCurrentLanguage = GamePreferences.getInstance(this).getGameLanguage();

        MaterialToolbar toolbar = findViewById(R.id.toolbar_home);
        setSupportActionBar(toolbar);
        if (getSupportActionBar() != null) {
            getSupportActionBar().setTitle(R.string.game_title);
            getSupportActionBar().setSubtitle(null);
        }

        mPanelTitle = findViewById(R.id.panel_title);
        mLayoutButtons = findViewById(R.id.layout_action_buttons);

        setupButtons();
        setupBottomNavigation();
        setupBackNavigation();
        playEntranceAnimations();
    }

    private void setupBackNavigation() {
        getOnBackPressedDispatcher().addCallback(this, new androidx.activity.OnBackPressedCallback(true) {
            @Override
            public void handleOnBackPressed() {
                showExitConfirmationDialog();
            }
        });
    }

    @Override
    protected void onResume() {
        super.onResume();
        UiUtils.applyImmersiveMode(this);
        String savedLang = GamePreferences.getInstance(this).getGameLanguage();
        if (mCurrentLanguage != null && !mCurrentLanguage.equals(savedLang)) {
            mCurrentLanguage = savedLang;
            recreate();
            return;
        }
        if (mBottomNav != null) {
            mBottomNav.setSelectedItemId(R.id.nav_home);
        }
    }

    private void playEntranceAnimations() {
        Animation slideLeft = AnimationUtils.loadAnimation(this, R.anim.slide_in_left);
        Animation slideRight = AnimationUtils.loadAnimation(this, R.anim.slide_in_right);

        if (mPanelTitle != null) {
            mPanelTitle.startAnimation(slideLeft);
        }
        if (mLayoutButtons != null) {
            mLayoutButtons.startAnimation(slideRight);
        }
    }

    private void setupButtons() {
        View btnPlay = findViewById(R.id.btn_play);
        View btnAchievements = findViewById(R.id.btn_achievements);
        View btnSettings = findViewById(R.id.btn_settings);
        View btnHowToPlay = findViewById(R.id.btn_how_to_play);
        View btnExit = findViewById(R.id.btn_exit);

        btnPlay.setOnClickListener(v -> UiUtils.animateClick(v, () -> {
            Intent intent = new Intent(HomeActivity.this, GameModeActivity.class);
            startActivity(intent);
            overridePendingTransition(R.anim.slide_in_right, R.anim.slide_out_left);
        }));

        btnAchievements.setOnClickListener(v -> UiUtils.animateClick(v, () -> {
            Intent intent = new Intent(HomeActivity.this, AchievementsActivity.class);
            startActivity(intent);
            overridePendingTransition(R.anim.slide_in_right, R.anim.slide_out_left);
        }));

        btnSettings.setOnClickListener(v -> UiUtils.animateClick(v, () -> {
            Intent intent = new Intent(HomeActivity.this, SettingsActivity.class);
            startActivity(intent);
            overridePendingTransition(R.anim.slide_in_right, R.anim.slide_out_left);
        }));

        btnHowToPlay.setOnClickListener(v -> UiUtils.animateClick(v, () -> {
            Intent intent = new Intent(HomeActivity.this, HowToPlayActivity.class);
            startActivity(intent);
            overridePendingTransition(R.anim.slide_in_right, R.anim.slide_out_left);
        }));

        btnExit.setOnClickListener(v -> UiUtils.animateClick(v, this::showExitConfirmationDialog));
    }

    private void setupBottomNavigation() {
        mBottomNav = findViewById(R.id.bottom_navigation);
        mBottomNav.setOnItemSelectedListener(item -> {
            int itemId = item.getItemId();
            if (itemId == R.id.nav_home) {
                return true;
            } else if (itemId == R.id.nav_achievements) {
                Intent intent = new Intent(HomeActivity.this, AchievementsActivity.class);
                startActivity(intent);
                overridePendingTransition(R.anim.fade_in, R.anim.fade_out);
                return true;
            } else if (itemId == R.id.nav_settings) {
                Intent intent = new Intent(HomeActivity.this, SettingsActivity.class);
                startActivity(intent);
                overridePendingTransition(R.anim.fade_in, R.anim.fade_out);
                return true;
            }
            return false;
        });
    }

    private void showExitConfirmationDialog() {
        com.example.monstersurvivor.android.audio.NativeAudio.play(this, com.example.monstersurvivor.audio.AudioEvent.UI_CANCEL);
        AlertDialog dialog = new AlertDialog.Builder(this)
                .setTitle(R.string.dialog_exit_title)
                .setMessage(R.string.dialog_exit_msg)
                .setIcon(R.drawable.ic_exit)
                .setPositiveButton(R.string.dialog_yes, (d, which) -> {
                    d.dismiss();
                    com.example.monstersurvivor.android.audio.NativeAudio.pause(this);
                    finishAffinity();
                    if (android.os.Build.VERSION.SDK_INT >= android.os.Build.VERSION_CODES.LOLLIPOP) {
                        finishAndRemoveTask();
                    }
                })
                .setNegativeButton(R.string.dialog_no, (d, which) -> {
                    com.example.monstersurvivor.android.audio.NativeAudio.play(this, com.example.monstersurvivor.audio.AudioEvent.UI_CANCEL);
                    d.dismiss();
                })
                .setOnCancelListener(d -> {
                    com.example.monstersurvivor.android.audio.NativeAudio.play(this, com.example.monstersurvivor.audio.AudioEvent.UI_CANCEL);
                })
                .create();
        dialog.show();
        UiUtils.styleDialog(dialog);
    }

    private void showAboutDialog() {
        AlertDialog dialog = new AlertDialog.Builder(this)
                .setTitle(R.string.dialog_about_title)
                .setMessage(R.string.dialog_about_msg)
                .setIcon(R.drawable.ic_info)
                .setPositiveButton(R.string.dialog_close, (d, which) -> d.dismiss())
                .create();
        dialog.show();
        UiUtils.styleDialog(dialog);
    }

    @Override
    public boolean onCreateOptionsMenu(Menu menu) {
        getMenuInflater().inflate(R.menu.menu_options_main, menu);
        return true;
    }

    @Override
    public boolean onOptionsItemSelected(@NonNull MenuItem item) {
        int id = item.getItemId();
        if (id == R.id.action_settings) {
            startActivity(new Intent(this, SettingsActivity.class));
            overridePendingTransition(R.anim.slide_in_right, R.anim.slide_out_left);
            return true;
        } else if (id == R.id.action_how_to_play) {
            startActivity(new Intent(this, HowToPlayActivity.class));
            overridePendingTransition(R.anim.slide_in_right, R.anim.slide_out_left);
            return true;
        } else if (id == R.id.action_about) {
            showAboutDialog();
            return true;
        } else if (id == R.id.action_theme_forest) {
            GamePreferences.getInstance(this).setTheme(GamePreferences.THEME_FOREST);
            Toast.makeText(this, R.string.toast_theme_forest, Toast.LENGTH_SHORT).show();
            recreate();
            return true;
        } else if (id == R.id.action_theme_dark) {
            GamePreferences.getInstance(this).setTheme(GamePreferences.THEME_DARK);
            Toast.makeText(this, R.string.toast_theme_dark, Toast.LENGTH_SHORT).show();
            recreate();
            return true;
        }
        return super.onOptionsItemSelected(item);
    }
}

/**
 * Screen to configure game difficulty (EASY/HARD) and game mode (NORMAL/DEMO).
 * Launches AndroidLauncher with explicit intent extras.
 */
public class GameModeActivity extends NativeAudioActivity {

    public static final String EXTRA_DIFFICULTY = "EXTRA_DIFFICULTY";
    public static final String EXTRA_GAME_MODE = "EXTRA_GAME_MODE";

    public static final String DIFFICULTY_EASY = "EASY";
    public static final String DIFFICULTY_HARD = "HARD";
    public static final String MODE_NORMAL = "NORMAL";
    public static final String MODE_DEMO = "DEMO";
    public static final String MODE_ENDLESS = "ENDLESS";

    private String mSelectedDifficulty = DIFFICULTY_EASY;
    private String mSelectedMode = MODE_NORMAL;
    private boolean mIsLaunchingGame = false;

    private View mCardDiffEasy;
    private View mCardDiffHard;
    private View mCardModeNormal;
    private View mCardModeDemo;
    private View mCardModeEndless;

    private TextView mTvDiffEasyTitle;
    private TextView mTvDiffHardTitle;
    private TextView mTvModeNormalTitle;
    private TextView mTvModeDemoTitle;
    private TextView mTvModeEndlessTitle;

    private TextView mTvBadge;
    private TextView mTvDescription;

    @Override
    protected void attachBaseContext(Context newBase) {
        super.attachBaseContext(UiUtils.wrapLocale(newBase));
    }

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        UiUtils.applyAppTheme(this);
        super.onCreate(savedInstanceState);
        UiUtils.applyImmersiveMode(this);
        setContentView(R.layout.activity_game_mode);

        MaterialToolbar toolbar = findViewById(R.id.toolbar_mode);
        toolbar.setNavigationOnClickListener(v -> UiUtils.navigateBack(this));

        getOnBackPressedDispatcher().addCallback(this, new androidx.activity.OnBackPressedCallback(true) {
            @Override
            public void handleOnBackPressed() {
                UiUtils.navigateBack(GameModeActivity.this);
            }
        });

        initViews();
        setupListeners();
        updateSelectionUI();
    }

    @Override
    protected void onResume() {
        super.onResume();
        mIsLaunchingGame = false;
        UiUtils.applyImmersiveMode(this);
    }

    private void initViews() {
        mCardDiffEasy = findViewById(R.id.card_diff_easy);
        mCardDiffHard = findViewById(R.id.card_diff_hard);
        mCardModeNormal = findViewById(R.id.card_mode_normal);
        mCardModeDemo = findViewById(R.id.card_mode_demo);
        mCardModeEndless = findViewById(R.id.card_mode_endless);

        mTvDiffEasyTitle = findViewById(R.id.tv_diff_easy_title);
        mTvDiffHardTitle = findViewById(R.id.tv_diff_hard_title);
        mTvModeNormalTitle = findViewById(R.id.tv_mode_normal_title);
        mTvModeDemoTitle = findViewById(R.id.tv_mode_demo_title);
        mTvModeEndlessTitle = findViewById(R.id.tv_mode_endless_title);

        mTvBadge = findViewById(R.id.tv_detail_selection_badge);
        mTvDescription = findViewById(R.id.tv_mode_description);
    }

    private void setupListeners() {
        mCardDiffEasy.setOnClickListener(v -> {
            if(DIFFICULTY_EASY.equals(mSelectedDifficulty))return;
            com.example.monstersurvivor.android.audio.NativeAudio.play(this,com.example.monstersurvivor.audio.AudioEvent.UI_NAVIGATION);
            mSelectedDifficulty = DIFFICULTY_EASY;
            updateSelectionUI();
        });

        mCardDiffHard.setOnClickListener(v -> {
            if(DIFFICULTY_HARD.equals(mSelectedDifficulty))return;
            com.example.monstersurvivor.android.audio.NativeAudio.play(this,com.example.monstersurvivor.audio.AudioEvent.UI_NAVIGATION);
            mSelectedDifficulty = DIFFICULTY_HARD;
            updateSelectionUI();
        });

        mCardModeNormal.setOnClickListener(v -> {
            if(MODE_NORMAL.equals(mSelectedMode))return;
            com.example.monstersurvivor.android.audio.NativeAudio.play(this,com.example.monstersurvivor.audio.AudioEvent.UI_NAVIGATION);
            mSelectedMode = MODE_NORMAL;
            updateSelectionUI();
        });

        mCardModeDemo.setOnClickListener(v -> {
            if(MODE_DEMO.equals(mSelectedMode))return;
            com.example.monstersurvivor.android.audio.NativeAudio.play(this,com.example.monstersurvivor.audio.AudioEvent.UI_NAVIGATION);
            mSelectedMode = MODE_DEMO;
            updateSelectionUI();
        });

        mCardModeEndless.setOnClickListener(v->{
            if(MODE_ENDLESS.equals(mSelectedMode))return;
            com.example.monstersurvivor.android.audio.NativeAudio.play(this,com.example.monstersurvivor.audio.AudioEvent.UI_NAVIGATION);
            mSelectedMode=MODE_ENDLESS;updateSelectionUI();
        });
        View btnStartGame = findViewById(R.id.btn_start_game);

        View btnPreview = findViewById(R.id.btn_preview_screens);
        if (btnPreview != null) {

            btnPreview.setVisibility(View.GONE);

    }

    private void updateSelectionUI() {
        // Update Difficulty cards
        boolean isEasy = DIFFICULTY_EASY.equals(mSelectedDifficulty);
        mCardDiffEasy.setBackgroundResource(isEasy ? R.drawable.panel_fantasy_card_selected : R.drawable.panel_fantasy_card);
        mTvDiffEasyTitle.setTextColor(ContextCompat.getColor(this, isEasy ? R.color.gold_bright : R.color.text_secondary));

        mCardDiffHard.setBackgroundResource(!isEasy ? R.drawable.panel_fantasy_card_selected : R.drawable.panel_fantasy_card);
        mTvDiffHardTitle.setTextColor(ContextCompat.getColor(this, !isEasy ? R.color.gold_bright : R.color.text_secondary));

        // Update Mode cards
        boolean isNormal = MODE_NORMAL.equals(mSelectedMode);
        boolean isDemo=MODE_DEMO.equals(mSelectedMode),isEndless=MODE_ENDLESS.equals(mSelectedMode);
        mCardModeNormal.setBackgroundResource(isNormal ? R.drawable.panel_fantasy_card_selected : R.drawable.panel_fantasy_card);
        mTvModeNormalTitle.setTextColor(ContextCompat.getColor(this, isNormal ? R.color.gold_bright : R.color.text_secondary));

        mCardModeDemo.setBackgroundResource(isDemo ? R.drawable.panel_fantasy_card_selected : R.drawable.panel_fantasy_card);
        mTvModeDemoTitle.setTextColor(ContextCompat.getColor(this, isDemo ? R.color.gold_bright : R.color.text_secondary));
        mCardModeEndless.setBackgroundResource(isEndless ? R.drawable.panel_fantasy_card_selected : R.drawable.panel_fantasy_card);
        mTvModeEndlessTitle.setTextColor(ContextCompat.getColor(this,isEndless?R.color.gold_bright:R.color.text_secondary));

        // Update Summary badge & description
        String diffText = isEasy ? getString(R.string.difficulty_easy) : getString(R.string.difficulty_hard);
        String modeText = getString(isEndless?R.string.mode_endless:isNormal?R.string.mode_normal:R.string.mode_demo);
        String badgeText = getString(R.string.mode_badge_format, diffText, modeText);
        mTvBadge.setText(badgeText);

        if(isEndless){
            mTvDescription.setText(R.string.desc_mode_endless);
        } else if (isNormal) {
            mTvDescription.setText(R.string.desc_mode_normal);
        } else {
            mTvDescription.setText(R.string.desc_mode_demo);
        }
    }

    private void launchGame() {
        if (mIsLaunchingGame) return;
        mIsLaunchingGame = true;
        Intent intent = new Intent(this, AndroidLauncher.class);
        intent.putExtra(EXTRA_DIFFICULTY, mSelectedDifficulty);
        intent.putExtra(EXTRA_GAME_MODE, mSelectedMode);
        startActivity(intent);
        overridePendingTransition(R.anim.fade_in, R.anim.fade_out);
    }

        mSwitchMusic.setChecked(mPrefs.isMusicEnabled());
        mSwitchSfx.setChecked(mPrefs.isSfxEnabled());
        mSwitchVibration.setChecked(mPrefs.isVibrationEnabled());

        int musicVol = mPrefs.getMusicVolume();
        mSeekMusic.setProgress(musicVol);
        mTvMusicVol.setText(getString(R.string.setting_music_volume, musicVol));

        int sfxVol = mPrefs.getSfxVolume();
        mSeekSfx.setProgress(sfxVol);
        mTvSfxVol.setText(getString(R.string.setting_sfx_volume, sfxVol));

        if ("en".equalsIgnoreCase(mPrefs.getGameLanguage())) {
            mRbLangEn.setChecked(true);
        } else {
            mRbLangVi.setChecked(true);
        }

        if (GamePreferences.THEME_DARK.equals(mPrefs.getTheme())) {
            mRbDark.setChecked(true);
        } else {
            mRbForest.setChecked(true);
        }

    private void setupListeners() {
        mSwitchMusic.setOnCheckedChangeListener((buttonView, isChecked) -> {
            mPrefs.setMusicEnabled(isChecked);
            com.example.monstersurvivor.android.audio.NativeAudio.refresh(this);
            mSeekMusic.setEnabled(isChecked);
        });

        mSwitchSfx.setOnCheckedChangeListener((buttonView, isChecked) -> {

            com.example.monstersurvivor.android.audio.NativeAudio.refresh(this);
            mSeekSfx.setEnabled(isChecked);

        mSwitchVibration.setOnCheckedChangeListener((buttonView, isChecked) ->
                mPrefs.setVibrationEnabled(isChecked));

            @Override
            public void onProgressChanged(SeekBar seekBar, int progress, boolean fromUser) {
                mTvMusicVol.setText(getString(R.string.setting_music_volume, progress));
                if (fromUser) {
                    mPrefs.setMusicVolume(progress);
                    com.example.monstersurvivor.android.audio.NativeAudio.refresh(SettingsActivity.this);
                }
            }

            @Override
            public void onStartTrackingTouch(SeekBar seekBar) {}

            @Override
            public void onStopTrackingTouch(SeekBar seekBar) {com.example.monstersurvivor.android.audio.NativeAudio.play(SettingsActivity.this,com.example.monstersurvivor.audio.AudioEvent.UI_CONFIRM);}
        });

        mSeekSfx.setOnSeekBarChangeListener(new SeekBar.OnSeekBarChangeListener() {
            @Override
            public void onProgressChanged(SeekBar seekBar, int progress, boolean fromUser) {
                mTvSfxVol.setText(getString(R.string.setting_sfx_volume, progress));
                if (fromUser) {
                    mPrefs.setSfxVolume(progress);
                    com.example.monstersurvivor.android.audio.NativeAudio.refresh(SettingsActivity.this);
                }
            }

            @Override
            public void onStartTrackingTouch(SeekBar seekBar) {}

            @Override
            public void onStopTrackingTouch(SeekBar seekBar) {com.example.monstersurvivor.android.audio.NativeAudio.play(SettingsActivity.this,com.example.monstersurvivor.audio.AudioEvent.UI_CONFIRM);}
        });

        mRgLanguage.setOnCheckedChangeListener((group, checkedId) -> {
            String currentLang = mPrefs.getGameLanguage();
            String targetLang = (checkedId == R.id.rb_lang_en) ? "en" : "vi";

            if (!targetLang.equalsIgnoreCase(currentLang)) {
                mPrefs.setGameLanguage(targetLang);
                Toast.makeText(this, R.string.toast_language_changed, Toast.LENGTH_SHORT).show();
                recreate();
            }
        });

        mRgTheme.setOnCheckedChangeListener((group, checkedId) -> {
            String currentTheme = mPrefs.getTheme();
            String targetTheme = (checkedId == R.id.rb_theme_dark)
                    ? GamePreferences.THEME_DARK
                    : GamePreferences.THEME_FOREST;

            if (!targetTheme.equals(currentTheme)) {
                mPrefs.setTheme(targetTheme);
                Toast.makeText(this, R.string.toast_theme_changed, Toast.LENGTH_SHORT).show();
                recreate();
            }
        });

        findViewById(R.id.btn_reset_defaults).setOnClickListener(v -> UiUtils.animateClick(v, () -> {
            mPrefs.resetDefaults();
            com.example.monstersurvivor.android.audio.NativeAudio.refresh(this);

            Toast.makeText(this, R.string.toast_defaults_reset, Toast.LENGTH_SHORT).show();
            recreate();
        }));

        AlertDialog dialog = new AlertDialog.Builder(this)
                .setTitle(R.string.dialog_about_title)
                .setMessage(R.string.dialog_about_msg)
                .setIcon(R.drawable.ic_info)
                .setPositiveButton(R.string.dialog_close, (d, which) -> d.dismiss())
                .create();
        dialog.show();
        UiUtils.styleDialog(dialog);
    }
}

    private void bindData() {
        // 1. Progress Summary Header
        int unlocked = mManager.getUnlockedAchievementsCount();
        int total = mManager.getTotalAchievementsCount();
        int percent = total > 0 ? (unlocked * 100 / total) : 0;
        if (mTvProgressSummary != null) {
            mTvProgressSummary.setText(getString(R.string.ach_progress_summary, unlocked, total, percent));
        }
        if (mProgressBar != null) {
            mProgressBar.setMax(total);
            mProgressBar.setProgress(unlocked);
        }

        // 2. Empty State Banner
        if (mPanelEmptyState != null) {
            mPanelEmptyState.setVisibility(mManager.hasAnyRecordedMatches() ? View.GONE : View.VISIBLE);
        }

        // 3. Lifetime Records
        TextView tvBestTime = findViewById(R.id.tv_val_best_time);
        TextView tvKills = findViewById(R.id.tv_val_total_kills);
        TextView tvVictories = findViewById(R.id.tv_val_victories);
        TextView tvMatches = findViewById(R.id.tv_val_matches);
        TextView tvBoss = findViewById(R.id.tv_val_boss);

        int bestSec = mManager.getBestSurvivalSeconds();
        long kills = mManager.getTotalKills();
        long victories = mManager.getTotalVictories();
        long matches = mManager.getTotalMatches();
        long bossDefeated = mManager.getBossDefeatedCount();

        if (tvBestTime != null) tvBestTime.setText(AchievementManager.formatTime(bestSec));
        if (tvKills != null) tvKills.setText(String.valueOf(kills));
        if (tvVictories != null) tvVictories.setText(String.valueOf(victories));
        if (tvMatches != null) tvMatches.setText(String.valueOf(matches));
        if (tvBoss != null) {
            if (bossDefeated > 0) {
                tvBoss.setText(getString(R.string.ach_boss_status_defeated, bossDefeated));
            } else {
                tvBoss.setText(R.string.ach_boss_status_none);
            }
        }

        // 4. Hero Achievements List
        List<AchievementManager.AchievementItem> items = mManager.getAchievements();
        for(int i=0;i<Math.min(items.size(),ACHIEVEMENT_VIEWS.length);i++) {
            int[] ids=ACHIEVEMENT_VIEWS[i];
            bindAchievementCard(findViewById(ids[0]),ids[1],ids[2],ids[3],ids[4],ids[5],items.get(i));
        }
        TextView extra=findViewById(R.id.tv_ach_extra_stats);
        extra.setText(getString(R.string.ach_extra_stats,AchievementManager.formatTime(mManager.getTotalPlaySeconds()),
            mManager.getMiniBossDefeatedCount(),mManager.getHardVictories(),mManager.getWeaponEvolutions()));
    }


    private void bindAchievementCard(View cardView, int iconId, int titleId, int descId, int progId, int badgeId,
                                     AchievementManager.AchievementItem item) {
        if (cardView == null) return;
        cardView.setTag(item);

        cardView.setBackgroundResource(item.isUnlocked
                ? R.drawable.panel_fantasy_card_selected
                : R.drawable.panel_fantasy_card);

        ImageView ivIcon = cardView.findViewById(iconId);
        if (ivIcon != null) {
            ivIcon.setImageResource(item.iconResId);
            ivIcon.setColorFilter(ContextCompat.getColor(this, item.isUnlocked
                    ? R.color.gold_bright
                    : R.color.text_muted));
        }

        TextView tvTitle = cardView.findViewById(titleId);
        if (tvTitle != null) {
            tvTitle.setText(item.titleResId);
            tvTitle.setTextColor(ContextCompat.getColor(this, item.isUnlocked
                    ? R.color.gold_bright
                    : R.color.text_secondary));
        }

        TextView tvDesc = cardView.findViewById(descId);
        if (tvDesc != null) {
            tvDesc.setText(item.descResId);
        }

        TextView tvProg = cardView.findViewById(progId);
        if (tvProg != null) {
            tvProg.setText(item.progressText+(item.isUnlocked?"\n"+unlockDate(item):""));
            tvProg.setTextColor(ContextCompat.getColor(this, item.isUnlocked
                    ? R.color.gold_text
                    : R.color.text_muted));
        }

        TextView tvBadge = cardView.findViewById(badgeId);
        if (tvBadge != null) {
            tvBadge.setText(item.isUnlocked ? R.string.ach_status_unlocked : R.string.ach_status_locked);
            tvBadge.setBackgroundResource(item.isUnlocked
                    ? R.drawable.panel_fantasy_badge_unlocked
                    : R.drawable.panel_fantasy_badge_locked);
            tvBadge.setTextColor(ContextCompat.getColor(this, item.isUnlocked
                    ? R.color.gold_bright
                    : R.color.text_muted));
        }
    }

        return getString(item.descResId)+"\n\n"+item.progressText+"\n"
            +getString(item.isUnlocked?R.string.ach_status_unlocked:R.string.ach_status_locked)
            +(item.isUnlocked?"\n"+unlockDate(item):"");

    private void setupBottomNavigation() {
        mBottomNav = findViewById(R.id.bottom_navigation);
        mBottomNav.setSelectedItemId(R.id.nav_achievements);
        mBottomNav.setOnItemSelectedListener(item -> {
            int itemId = item.getItemId();
            if (itemId == R.id.nav_home) {
                Intent intent = new Intent(AchievementsActivity.this, HomeActivity.class);
                intent.setFlags(Intent.FLAG_ACTIVITY_CLEAR_TOP | Intent.FLAG_ACTIVITY_SINGLE_TOP);
                startActivity(intent);
                overridePendingTransition(R.anim.fade_in, R.anim.fade_out);
                finish();
                return true;
            } else if (itemId == R.id.nav_achievements) {
                return true;
            } else if (itemId == R.id.nav_settings) {
                Intent intent = new Intent(AchievementsActivity.this, SettingsActivity.class);
                startActivity(intent);
                overridePendingTransition(R.anim.fade_in, R.anim.fade_out);
                return true;
            }
            return false;
        });

    @Override
    public void onCreateContextMenu(ContextMenu menu, View v, ContextMenu.ContextMenuInfo menuInfo) {

        mSelectedCardId = v.getId();
        getMenuInflater().inflate(R.menu.menu_context_achievement, menu);

        if (mSelectedCardId == R.id.card_ach_best_time) {
            menu.setHeaderTitle(R.string.ach_best_time);
        } else if (mSelectedCardId == R.id.card_ach_kills) {
            menu.setHeaderTitle(R.string.ach_total_kills);
        } else if (mSelectedCardId == R.id.card_ach_victories) {
            menu.setHeaderTitle(R.string.ach_victories);
        } else if (mSelectedCardId == R.id.card_ach_matches) {
            menu.setHeaderTitle(R.string.ach_matches);
        } else if (mSelectedCardId == R.id.card_ach_boss) {
            menu.setHeaderTitle(R.string.ach_boss);
        } else {
            var selected=achievement(mSelectedCardId);
            if(selected!=null)menu.setHeaderTitle(selected.titleResId);
        }

    @Override
    public boolean onContextItemSelected(@NonNull MenuItem item) {
        int itemId = item.getItemId();
        if (itemId == R.id.context_view_details) {
            showAchievementDetails(mSelectedCardId);
            return true;
        } else if (itemId == R.id.context_copy_info) {
            copyAchievementInfo(mSelectedCardId);
            return true;
        } else if (itemId == R.id.context_share_info) {
            shareAchievementInfo(mSelectedCardId);
            return true;
        }
        return super.onContextItemSelected(item);
    }

    private void showAchievementDetails(int cardId) {
        String title;
        String message;

            message = getString(R.string.ach_best_time_detail,
                    getString(R.string.ach_best_time_desc),
                    AchievementManager.formatTime(mManager.getBestSurvivalSeconds()));
        } else if (cardId == R.id.card_ach_kills) {
            title = getString(R.string.ach_total_kills);

public final class AchievementManager {
    private static final String PREF_NAME="monster_survivor_stats";
    // Previous UI placeholders had no verified mode/source. Never import them as official results.
    private static final String PREFIX="official_v1.";

    public static synchronized AchievementManager getInstance(Context context) {

    public int getBestSurvivalSeconds() {return (int)value(KEY_BEST_SURVIVAL_SECONDS);}

        public final String id;
        public final int titleResId,descResId,iconResId;

        private AchievementItem(String id,int title,int desc,int icon,long current,long target,long achieved) {
            this.id=id;titleResId=title;descResId=desc;iconResId=icon;

        add(list,"hard_victor",R.string.ach_item_hard_title,R.string.ach_item_hard_desc,R.drawable.ic_shield,hard,1);
        add(list,"weapon_master",R.string.ach_item_evolution_title,R.string.ach_item_evolution_desc,R.drawable.ic_sparkles,evolutions,1);
        return list;
    }
    public synchronized List<AchievementItem> getAchievements() {
        return catalog(getTotalMatches(),getTotalKills(),getBestSurvivalSeconds(),getMiniBossDefeatedCount(),getTotalVictories(),getHardVictories(),getWeaponEvolutions());
    }
    public int getUnlockedAchievementsCount() {int n=0;for(var item:getAchievements())if(item.isUnlocked)n++;return n;}
    public int getTotalAchievementsCount() {return 8;}
    public static String formatTime(long totalSeconds) {
        return String.format(Locale.getDefault(),"%02d:%02d",totalSeconds/60,totalSeconds%60);
    }

    /** One atomic disk commit includes counters, unlock dates and the durable idempotency ledger. */
    public synchronized boolean recordMatchResult(MatchResult result) {
        if(result==null || result.mode()!=GameMode.NORMAL || result.diagnostic() || result.difficulty()==null
            || result.matchId()==null || !result.matchId().matches("[a-f0-9]{8}(-[a-f0-9]{4}){3}-[a-f0-9]{12}")
            || !Float.isFinite(result.survivedSeconds()) || result.survivedSeconds()<0
            || !Float.isFinite(result.progressSeconds()) || result.progressSeconds()<0
            || result.kills()<0 || result.weaponEvolutions()<0 || result.weaponEvolutions()>4
            || (result.victory() && (!result.finalBossDefeated() || result.progressSeconds()<600)))return false;
        var recorded=new HashSet<>(mPrefs.getStringSet(RECORDED,java.util.Collections.emptySet()));
        if(!recorded.add(result.matchId()))return false;
        long matches=getTotalMatches()+1,kills=getTotalKills()+result.kills();
        long best=Math.max(getBestSurvivalSeconds(),(int)Math.min(Integer.MAX_VALUE,result.survivedSeconds()));
        long mini=getMiniBossDefeatedCount()+(result.miniBossDefeated()?1:0);
        long lich=getBossDefeatedCount()+(result.finalBossDefeated()?1:0);
        long victories=getTotalVictories()+(result.victory()?1:0);
        long hard=getHardVictories()+(result.victory() && result.difficulty()==Difficulty.HARD?1:0);
        long evolutions=getWeaponEvolutions()+result.weaponEvolutions();
        long play=value(PLAY_MS)+(long)(result.survivedSeconds()*1000d);
        var editor=mPrefs.edit().putLong(KEY_TOTAL_MATCHES,matches).putLong(KEY_TOTAL_KILLS,kills)
            .putLong(KEY_BEST_SURVIVAL_SECONDS,best).putLong(MINI,mini).putLong(KEY_BOSS_DEFEATED_COUNT,lich)
            .putLong(KEY_TOTAL_VICTORIES,victories).putLong(HARD,hard).putLong(EVOLUTIONS,evolutions)
            .putLong(PLAY_MS,play).putStringSet(RECORDED,recorded);
        long now=System.currentTimeMillis();
        for(var item:catalog(matches,kills,best,mini,victories,hard,evolutions)) {
            if(!item.isUnlocked && item.currentProgress>=item.targetProgress)editor.putLong(PREFIX+"unlocked."+item.id,now);
        }
        boolean saved=editor.commit();
        if(!saved)Log.e("Achievements","Could not persist completed match "+result.matchId());
        return saved;
    }
}

    private GameScreen(GameConfig config,Runnable onHome,int exhibitCount,DiagnosticRun diagnostics) {
        this.diagnostics=diagnostics;
        // Exactly one tuning fixture per explicit run. Legacy files at the local root are never read.
        skillDiagnostics=com.example.monstersurvivor.debug.SkillDiagnostics.load(diagnostics);
        bossDiagnostics=skillDiagnostics==null?com.example.monstersurvivor.debug.BossDiagnostics.load(diagnostics):null;
        profile=skillDiagnostics==null && bossDiagnostics==null?CombatProfile.load(diagnostics):null;
        trace=skillDiagnostics==null && bossDiagnostics==null && profile==null?CameraTrace.load(diagnostics):null;
        view=new PixelCamera(trace==null?PixelCamera.Mode.SCREEN:trace.mode);
        this.config=skillDiagnostics!=null?skillDiagnostics.tuning(config):bossDiagnostics!=null?bossDiagnostics.tuning(config):profile==null?config:profile.tuning(config);this.onHome=onHome;
        hud=new GameHud(this.config.strings,assets);input=new GameInput(hud,this);
        // Existing local camera measurement explicitly opts into the isolated preview scene.
        exhibitMode=exhibitCount>0 || trace!=null;
        fixedMapSeed=debugSeed(this.config.fixedMapSeed,diagnostics);
        map=exhibitMode || (profile!=null && !profile.worldProbe)?new ForestMap(assets):new ForestMap(assets,nextSeed());
        viewport=new ExtendViewport(320/this.config.visuals.zoom,240/this.config.visuals.zoom,map.pixelWidth(),map.pixelHeight(),camera);
        playerSprite=new WorldSprite(playerSet.frame(State.IDLE,Direction.DOWN,0),map.spawnX,map.spawnY,
            playerSet.feetX,playerSet.height-playerSet.feetY,playerSet.displayScale);
        session=new GameSession(this.config,map.grid,map.spawnX,map.spawnY,assets,skillDiagnostics!=null?skillDiagnostics.metrics:bossDiagnostics!=null?bossDiagnostics.metrics:profile==null?null:profile.metrics,map.seed);player=session.player;
        session.audio(audio);audio.listener(player.x,player.y);
        hud.setDebug(debugEnabled());
        if(exhibitMode) {
            exhibit("tiny_slime",-78,38); exhibit("tiny_bat",-26,52); exhibit("giant_bee",30,52);
            exhibit("red_cap",77,38); exhibit("living_armor",-65,-42); exhibit("lich_lord",65,-50);
            String[] ids={"honey_slime","wild_boar","giant_bee","red_cap","red_crab","giant_frog","rock_golem","living_armor","bone_hound"};
            for (int i=6;i<exhibitCount;i++) exhibit(ids[i%ids.length],(i%10-4.5f)*26,(i/10-2.5f)*26);
        }
        Gdx.app.log("MonsterSurvivor", "Ready: "+this.config.difficulty+" "+this.config.mode+" debug="+debugEnabled()+" preview="+exhibitMode
            +" seed="+map.seed+" walkable="+map.walkableTiles+" fallback="+map.fallback+" sword="+this.config.swordRange+"/"+this.config.swordCooldown
            +" roster="+com.example.monstersurvivor.enemy.EnemyRegistry.all().size()+" textures="+assets.textureCount()+" rgbaBytes="+assets.textureBytes()+" language="+this.config.language.code
            +" initialEnemies="+this.config.initialEnemies+" residentCap="+this.config.maxEnemies+" enemySpeedMultiplier="+this.config.enemySpeedMultiplier);

    }

    private static Long debugSeed(Long configured,DiagnosticRun diagnostics) {
        if(configured!=null)return configured;
        if(!diagnostics.enabled())return null;
        var file=diagnostics.file("map-seed.txt");if(!file.exists())return null;
        try {return Long.valueOf(file.readString("UTF-8").trim());}
        catch(NumberFormatException error) {Gdx.app.log("MonsterSurvivor","Invalid map-seed.txt; using random seed");return null;}
    }
    private long nextSeed() {
        if(fixedMapSeed!=null)return fixedMapSeed;
        long seed=java.util.concurrent.ThreadLocalRandom.current().nextLong();return map!=null && seed==map.seed?seed+1:seed;
    }

    @Override public void render(float delta) {
        syncAudio();
        if(profile!=null)profile.beforeFrame(this);
        if(bossDiagnostics!=null)bossDiagnostics.before(this);
        if(skillDiagnostics!=null)skillDiagnostics.before(this);
        boolean wasGameOver=gameOver();
        if(trace!=null)trace.beforeFrame(this);
        float step = dropNextDelta ? 0 : Math.min(Math.max(delta,0),.1f);
        dropNextDelta = false;
        input.sampleAxes();
        if (exhibitMode && !isPaused()) {
            elapsed += step; exhibitTime += step;
            player.update(step,input.axisX(),input.axisY(),map.grid);
        } else if(!exhibitMode) {
            // Follow before spawn sampling so the exclusion rectangle matches the current camera.
            followCamera();
            session.update(step,input.axisX(),input.axisY(),isPaused(),
                camera.position.x-camera.viewportWidth/2,camera.position.y-camera.viewportHeight/2,
                camera.position.x+camera.viewportWidth/2,camera.position.y+camera.viewportHeight/2);
            elapsed=session.elapsed();syncAudio();
        }
        if(session.choosingWeapon() && session.selection.token()!=loggedSelectionToken) {
            loggedSelectionToken=session.selection.token();input.reset();dropNextDelta=true;
            StringBuilder choices=new StringBuilder();for(int i=0;i<session.selection.size();i++) {
                if(i>0)choices.append('|');choices.append(session.selection.get(i).title);
            }
            Gdx.app.log("MonsterSurvivor","Weapon choice id="+session.selection.pendingId()+" token="+session.selection.token()
                +" time="+elapsed+" player="+player.x+","+player.y+" offers="+choices+" loadout="+session.weapons.loadout.describe());
            StringBuilder types=new StringBuilder();for(int i=0;i<session.selection.size();i++) {if(i>0)types.append('|');var offer=session.selection.get(i);types.append(offer.weapon).append(':').append(offer.action()).append(':').append(offer.fromLevel).append("->").append(offer.toLevel);}
            Gdx.app.log("WeaponRewards","Choice token="+session.selection.token()+" offers="+types);
        }
        if(session.choosingSkill() && session.skillSelection.token()!=loggedSkillSelectionToken) {
            loggedSkillSelectionToken=session.skillSelection.token();input.reset();dropNextDelta=true;
            StringBuilder choices=new StringBuilder();for(int i=0;i<session.skillSelection.size();i++) {
                if(i>0)choices.append('|');var offer=session.skillSelection.get(i);choices.append(offer.type).append(':').append(offer.fromLevel).append("->").append(offer.toLevel);
            }
            Gdx.app.log("Skills","Choice id="+session.skillSelection.pendingId()+" token="+session.skillSelection.token()+" offers="+choices+" elapsed="+session.elapsed()+" progress="+session.progress());
        }
        if(loggedSkillBoxes!=session.pickups.spawnedSkillBoxes) {
            loggedSkillBoxes=session.pickups.spawnedSkillBoxes;
            for(int i=0;i<session.pickups.size();i++) {var p=session.pickups.get(i);if(p.type==com.example.monstersurvivor.pickup.PickupType.SKILL_BOX)
                Gdx.app.log("Skills","Box id="+p.id+" player="+player.x+","+player.y+" item="+p.x+","+p.y+" elapsed="+session.elapsed()+" progress="+session.progress()+" count="+session.pickups.skillBoxes());}
        }
        if(!wasGameOver && gameOver())Gdx.app.log("MonsterSurvivor","Game Over seed="+map.seed+" survived="+elapsed+" kills="+session.kills);
        if(session.pickups.noticeSerial!=loggedNotice) {
            loggedNotice=session.pickups.noticeSerial;
            Gdx.app.log("MonsterSurvivor","Pickup "+session.pickups.notice+" hp="+player.hp()+"/"+player.maxHp()
                +" dmg="+player.stats.damage()+" speed="+player.stats.speed()+" collected="+session.pickups.collectedItems);
        }
        if((config.debug || fixedMapSeed!=null) && session.pickups.spawnedBoxes!=loggedBoxes) {
            loggedBoxes=session.pickups.spawnedBoxes;
            for(int i=0;i<session.pickups.size();i++) {
                var item=session.pickups.get(i);
                if(item.type.box && item.age==0)Gdx.app.log("MonsterSurvivor","Box "+item.type+" at="+item.x+","+item.y+" time="+elapsed+" progress="+session.progress());
            }
        }
        followCamera();
        FrameMetrics metrics=session.metrics;
        long worldStart=metrics==null?0:metrics.start(),animationBefore=metrics==null?0:metrics.value(ANIMATION);
        ScreenUtils.clear(.17f,.23f,.13f,1);
        viewport.apply();
        batch.setProjectionMatrix(camera.combined);
        batch.begin();
        map.renderGround(batch,camera);
        visible.clear();
        float left=camera.position.x-camera.viewportWidth/2, bottom=camera.position.y-camera.viewportHeight/2;
        float right=left+camera.viewportWidth, top=bottom+camera.viewportHeight;
        map.enqueueProps(visible,left,bottom,right,top);
        if(!exhibitMode)pickupRenderer.enqueue(session.pickups,visible,left,bottom,right,top);
        playerSprite.x=view.playerX; playerSprite.y=view.playerY;
        // Preview buttons affect the stationary Warrior too. Movement always keeps its original Walk logic.
        boolean previewPlayer=exhibitMode && !player.walking && stateIndex>0;
        long animationStart=metrics==null?0:metrics.start();
        State playerState=exhibitMode?(player.walking?State.WALK:PREVIEW_STATES[previewPlayer?stateIndex:0]):player.state();
        playerSprite.frame=playerSet.frame(playerState,
            previewPlayer?DIRECTIONS[directionIndex]:player.direction,previewPlayer?exhibitTime:player.renderAnimationTime());
        if(metrics!=null)metrics.finish(ANIMATION,animationStart);
        enqueue(playerSprite,left,bottom,right,top);
        for (int i=0;i<exhibits.size;i++) {
            Exhibit e=exhibits.get(i);
            e.sprite.frame=e.set.frame(PREVIEW_STATES[stateIndex],DIRECTIONS[directionIndex],exhibitTime);
            enqueue(e.sprite,left,bottom,right,top);
        }
        for(int i=0;i<session.enemies.size();i++) {
            Enemy e=session.enemies.get(i);
            if(e.definition.index>=0 && (config.debug || fixedMapSeed!=null) && !loggedSpecies[e.definition.index]) {
                loggedSpecies[e.definition.index]=true;
                Gdx.app.log("MonsterSurvivor","Enemy species="+e.definition.id+" progress="+session.progress()+" hp="+e.stats.hp+" flying="+e.definition.flying);
            }
            if(e.x<left-80 || e.x>right+80 || e.y<bottom-80 || e.y>top+80)continue;
            animationStart=metrics==null?0:metrics.start();e.prepareSprite();
            if(metrics!=null)metrics.finish(ANIMATION,animationStart);
            enqueue(e.sprite,left,bottom,right,top);
        }
        visible.sort(WorldSprite.BACK_TO_FRONT);
        for (int i=0;i<visible.size;i++) {
            WorldSprite sprite=visible.get(i);
            if(sprite==playerSprite && player.alive() && player.immunity()>0) {
                float alpha=((int)(player.immunity()*16)&1)==0?.4f:1;
                batch.setColor(1,.55f,.55f,alpha);sprite.draw(batch);batch.setColor(1,1,1,1);
            } else sprite.draw(batch);
        }
        batch.end();
        worldDrawCalls=batch.renderCalls;
        if(worldStart!=0)metrics.add(WORLD_RENDER,System.nanoTime()-worldStart-(metrics.value(ANIMATION)-animationBefore));
        if(!exhibitMode) {
            long start=metrics==null?0:metrics.start();
            float nightStrength=profile!=null && profile.worldProbe?profile.nightStrength():session.night.strength();
            if(nightStrength>0) {
                nightRenderer.draw(batch,config.visuals,nightStrength,camera.viewportWidth,camera.viewportHeight,view.playerX-left,view.playerY-bottom);
                worldDrawCalls+=batch.renderCalls;
            }
            if(metrics!=null)metrics.finish(NIGHT_RENDER,start);
            start=metrics==null?0:metrics.start();batch.setProjectionMatrix(camera.combined);batch.begin();
            bossRenderer.draw(batch,session.bosses,left,bottom,right,top);
            weaponRenderer.draw(batch,session.weapons,left,bottom,right,top);
            skillRenderer.draw(batch,session.skills,player.x,player.y,left,bottom,right,top);batch.end();worldDrawCalls+=batch.renderCalls;
            if(metrics!=null)metrics.finish(WEAPON_RENDER,start);
            start=metrics==null?0:metrics.start();batch.begin();
            enemySkillRenderer.draw(batch,session.enemies,player,left,bottom,right,top);batch.end();worldDrawCalls+=batch.renderCalls;
            if(metrics!=null)metrics.finish(ENEMY_FX,start);
        }
        long hudStart=metrics==null?0:metrics.start();
        hud.render(batch,input,isPaused(),elapsed,Math.min(delta,.5f),PREVIEW_STATES[stateIndex],DIRECTIONS[directionIndex],
            player,session.kills,config.mode,gameOver(),session.pickups,camera,session);
        if(loggedBossNotice!=session.bosses.noticeSerial) {
            loggedBossNotice=session.bosses.noticeSerial;var b=session.bosses.primary();
            Gdx.app.log("MonsterSurvivor","Boss event="+session.bosses.noticeKey()+" progress="+session.progress()+" elapsed="+session.elapsed()
                +" mini="+session.bosses.miniState()+" final="+session.bosses.finalState()+" actor="+(b==null?"pending":b.definition.body().id+" hp="+b.hp()+"/"+b.maxHp()+" phase="+b.phase()+" xy="+b.actor.x+","+b.actor.y));
        }
        if(victory() && !loggedVictory) {loggedVictory=true;Gdx.app.log("MonsterSurvivor","Victory result="+session.result());}
        if(loggedSkillImpacts!=session.skills.ultimateImpacts) {
            loggedSkillImpacts=session.skills.ultimateImpacts;
            Gdx.app.log("Skills","Impact token="+session.skills.view(com.example.monstersurvivor.skill.SkillType.ULTIMATE).activationId()
                +" impacts="+loggedSkillImpacts+" hits="+session.skills.ultimateHits+" kills="+session.skills.ultimateKills+" knockbacks="+session.skills.knockbacks
                +" bossHits="+session.skills.ultimateBossHits
                +" elapsed="+session.elapsed()+" radius="+session.skills.ultimateRadius()+" phase="+session.phase());
        }
        if(metrics!=null)metrics.finish(HUD,hudStart);
        if(trace!=null)trace.afterFrame(this,view,delta,step);
        if(profile!=null)profile.afterFrame(this,delta);
        if(bossDiagnostics!=null)bossDiagnostics.after(this);
        if(skillDiagnostics!=null)skillDiagnostics.after(this);
    }

    }

    @Override public boolean chooseWeapon(int index,long token) {
        if(isPaused() || !weaponSelection())return false;
        if(session.choosingSkill()) {
            if(index<0 || index>=session.skillSelection.size())return false;
            var offer=session.skillSelection.get(index);
            if(!session.chooseSkill(index,token))return false;
            input.reset();dropNextDelta=true;
            Gdx.app.log("Skills","Upgraded type="+offer.type+" from="+offer.fromLevel+" to="+offer.toLevel+" token="+token
                +" dashCd="+session.skills.view(com.example.monstersurvivor.skill.SkillType.DASH).cooldownRemaining()
                +" ultCd="+session.skills.view(com.example.monstersurvivor.skill.SkillType.ULTIMATE).cooldownRemaining()
                +" weapons="+session.weapons.loadout.describe()+" elapsed="+session.elapsed());return true;
        }
        String title=index>=0 && index<session.selection.size()?session.selection.get(index).title:"invalid";
        if(!session.chooseWeapon(index,token))return false;
        input.reset();dropNextDelta=true;
        Gdx.app.log("MonsterSurvivor","Weapon selected "+title+" time="+elapsed+" loadout="+session.weapons.loadout.describe());return true;
    }
    @Override public void retry() {
        if(!gameOver() && !victory())return;
        audio.reset();session.end();
        if(!exhibitMode && (profile==null || profile.worldProbe))map=new ForestMap(assets,nextSeed());

        Gdx.app.log("MonsterSurvivor","Retry seed="+map.seed+" walkable="+map.walkableTiles+" fallback="+map.fallback);

    private boolean skillBox(Pickup item,Player p) {
        if(!p.alive())return false;
        if(skillSelection.pendingId()==item.id)return skillRewardCommitted;
        // Endless never leaves an unusable Skill Box after both skills reach Lv5.
        // Reuse the existing Weapon Box selection, including its valid stat fallbacks.
        if(endless() && !skills.hasUpgrade())return weaponBox(item,p);
        if(phase==Phase.RUNNING && skillSelection.open(item.id,skills))phase=Phase.CHOOSING_SKILL;
        return false;
    }
    public boolean chooseSkill(int index,long token) {
        if(!choosingSkill() || !player.alive() || !pickups.contains(skillSelection.pendingId()))return false;
        if(!skillSelection.commit(index,token,skills))return false;
        var offer=skillSelection.get(index);skillRewardCommitted=true;
        if(!pickups.collect(skillSelection.pendingId()))throw new IllegalStateException("Pending skill item lost");
        pickups.announce(PickupType.SKILL_BOX,offer.title);skillSelection.clear();skillRewardCommitted=false;phase=Phase.RUNNING;
        if(!skills.hasUpgrade() && !endless())pickups.retireSkillBoxes();
        return true;
    }
    private boolean weaponBox(Pickup item,Player p) {
        if(!p.alive())return false;
        if(selection.pendingId()==item.id)return rewardCommitted;
        if(phase==Phase.RUNNING && selection.open(item.id,weapons.loadout,p)) {
            phase=Phase.CHOOSING_WEAPON;audio.play(com.example.monstersurvivor.audio.AudioEvent.PICKUP_WEAPON_OPEN);
        }
        return false;
    }
    /** Token and pending identity make double taps, stale UI callbacks and repeated commits harmless. */
    public boolean chooseWeapon(int index,long token) {
        if(!choosingWeapon() || !player.alive() || token!=selection.token() || index<0 || index>=selection.size()
            || !pickups.contains(selection.pendingId()))return false;
        WeaponOffer offer=selection.get(index);if(!offer.apply(weapons.loadout,player))return false;
        rewardCommitted=true;
        if(!pickups.collect(selection.pendingId()))throw new IllegalStateException("Pending weapon item lost");
        pickups.announce(PickupType.WEAPON_BOX,offer.title);selection.clear();rewardCommitted=false;phase=Phase.RUNNING;
        if(offer.kind==WeaponOffer.Kind.EVOLUTION)weaponEvolutions++;
        audio.play(switch(offer.kind) {
            case WEAPON -> offer.fromLevel==0?com.example.monstersurvivor.audio.AudioEvent.PICKUP_WEAPON_SELECTED:com.example.monstersurvivor.audio.AudioEvent.PICKUP_WEAPON_UPGRADED;
            case EVOLUTION -> com.example.monstersurvivor.audio.AudioEvent.PICKUP_WEAPON_UPGRADED;
            case DAMAGE -> com.example.monstersurvivor.audio.AudioEvent.PICKUP_DAMAGE;
            case MAX_HP -> com.example.monstersurvivor.audio.AudioEvent.PICKUP_HEALTH;
            case SPEED -> com.example.monstersurvivor.audio.AudioEvent.PICKUP_SPEED;
        });
        player.limitSwordCooldown(weapons.swordCooldown());return true;
    }
    private void clearSelections() {selection.clear();skillSelection.clear();rewardCommitted=skillRewardCommitted=false;}
    public void end() {phase=Phase.ENDED;skills.cancel();player.statuses.clear();clearSelections();weapons.clearTransient();bosses.clear();enemies.clear();}
    public boolean victory() {return phase==Phase.VICTORY;}

        phase=won?Phase.VICTORY:Phase.GAME_OVER;

            matchId,diagnosticResult || config.debug,weaponEvolutions,bosses.endlessKills(),endless()?endlessStage():0);

    }

    public void update(float delta,float axisX,float axisY,boolean paused,float left,float bottom,float right,float top) {
        if(choosingReward() && !player.alive())dying();
        if(paused || phase==Phase.GAME_OVER || phase==Phase.VICTORY || phase==Phase.ENDED || choosingReward() || delta<=0)return;
        delta=Math.min(delta,.1f);
        enemies.beginFrame();
        // Movement already sweeps in <=3-unit substeps. Repeating all AI/LOS/separation
        // when a frame exceeds 33ms caused a load feedback loop on the emulator.
        float dt=delta;
        if(phase==Phase.WINNING) {
            enemies.advanceDeaths(dt);
            if(bosses.lich().actor.deathFinished())finish(true);
            return;
        }
        if(phase==Phase.RUNNING && !player.alive())dying();
        if(phase==Phase.DYING) {
            player.update(dt,0,0,grid);
            if(player.deathFinished())finish(false);
            return;
        }
        if(phase!=Phase.RUNNING)return;
        elapsed+=dt;progress+=dt*config.mode.progressScale;
        night.update(dt,progress());
        long start=metrics==null?0:metrics.start(),collisionBefore=metrics==null?0:metrics.value(COLLISION);
        skills.advance(dt,axisX,axisY,player,grid);
        player.statuses.update(dt,player);
        if(start!=0)metrics.add(PLAYER,System.nanoTime()-start-(metrics.value(COLLISION)-collisionBefore));
        if(!player.alive()){dying();return;}
        bosses.prepare(dt,progress(),player,left,bottom,right,top);
        start=metrics==null?0:metrics.start();
        enemies.skills.prepare(dt,player,progress(),left,bottom,right,top);
        if(metrics!=null)metrics.finish(ENEMY_SKILLS,start);
        enemies.update(dt,player);
        // Apply physical pickups before a pending sword hit reads the current match stats.
        start=metrics==null?0:metrics.start();collisionBefore=metrics==null?0:metrics.value(COLLISION);
        pickups.update(dt,progress(),player);
        if(start!=0)metrics.add(PICKUPS,System.nanoTime()-start-(metrics.value(COLLISION)-collisionBefore));
        if(choosingReward())return;
        start=metrics==null?0:metrics.start();collisionBefore=metrics==null?0:metrics.value(COLLISION);
        long projectileBefore=metrics==null?0:metrics.value(PROJECTILES);
        weapons.update(dt,player,enemies,grid);
        if(!bosses.finalDefeated())skills.resolve(player,enemies,grid,weapons.projectiles);
        if(!bosses.finalDefeated()){
            long enemyStart=metrics==null?0:metrics.start();enemies.skills.resolve(dt,player);
            if(metrics!=null)metrics.finish(ENEMY_PROJECTILES,enemyStart);
        }
        // A completed final kill freezes combat before any surviving enemy can hit.
        if(!bosses.finalDefeated())combat.contact(player,enemies);
        if(start!=0)metrics.add(COMBAT,System.nanoTime()-start-(metrics.value(COLLISION)-collisionBefore)-(metrics.value(PROJECTILES)-projectileBefore));
        if(!player.alive()) {dying();return;}
        if(!endless() && bosses.finalDefeated() && progress()>=config.bosses.finalProgress) {
            phase=Phase.WINNING;skills.cancel();player.statuses.clear();enemies.skills.clear();clearSelections();weapons.clearTransient();bosses.stopCombat();return;
        }
        bosses.fight(dt,progress(),player);
        if(!player.alive()) {dying();return;}
        start=metrics==null?0:metrics.start();collisionBefore=metrics==null?0:metrics.value(COLLISION);
        spawner.update(dt,progress(),player,left,bottom,right,top);
        if(start!=0)metrics.add(SPAWN,System.nanoTime()-start-(metrics.value(COLLISION)-collisionBefore));
    }

/** Pointer ownership allows joystick + pause/preview touches without stealing movement. */
public final class GameInput extends InputAdapter {
    /** HUD units, bottom-left origin after unproject; presentation may implement this surface. */
    public interface Surface {
        float width(); float height();
        void unproject(int x,int y,Vector2 out);
        boolean inPauseButton(float x,float y); boolean inResumeButton(float x,float y); boolean inHomeButton(float x,float y);
        boolean inAnimationButton(float x,float y); boolean inDirectionButton(float x,float y);

    }
    public interface Commands {
        boolean isPaused();
        void togglePause();
        void cycleAnimation();
        void cycleDirection();
        default boolean debugEnabled() { return false; }
        default boolean gameplayStopped() { return false; }
        default boolean gameOver() { return false; }
        default boolean victory() { return false; }
        default void retry() { }
        default void home() { }
        default boolean weaponSelection() {return false;}
        default long weaponToken() {return -1;}
        default int weaponOfferCount() {return 0;}

    }
    public static final float JOY_X = 108, JOY_Y = 108, JOY_RADIUS = 66;
    public static final float TOUCH_WIDTH_FRACTION=.44f, TOP_RESERVED=96, BOTTOM_RESERVED=28, DEADZONE=.12f;
    private final boolean[] keys = new boolean[256];
    private final Vector2 touch = new Vector2();
    private final Surface hud;
    private final Commands commands;
    private int pointer = -1;
    private int modalPointer=-1,selectedOffer=-1;
    private final boolean[] heldPointers=new boolean[32];
    private int dashPointer=-1,ultimatePointer=-1;
    private boolean leftShiftHeld,rightShiftHeld,spaceHeld;
    private long modalToken,selectionBlockedUntil;
    private float joyX, joyY, axisX, axisY;
    private float centerX=JOY_X,centerY=JOY_Y;
    public GameInput(Surface hud, Commands commands) { this.hud = hud; this.commands = commands; }
    private void releaseJoystick() {
        pointer=-1;joyX=joyY=axisX=axisY=0;centerX=JOY_X;centerY=JOY_Y;
    }
    private void resetMovement() {
        java.util.Arrays.fill(keys,false);releaseJoystick();
    }
    public void reset() {
        resetMovement();
        java.util.Arrays.fill(heldPointers,false);dashPointer=ultimatePointer=-1;
        modalPointer=-1;selectedOffer=-1;
    }
    public int selectedOffer() {return selectedOffer;}
    private boolean select(int index,long token) {
        long now=com.badlogic.gdx.utils.TimeUtils.nanoTime();
        if(now<selectionBlockedUntil)return false;
        boolean applied=commands.chooseWeapon(index,token);
        if(applied) {reset();selectionBlockedUntil=now+250_000_000L;}return applied;
    }
    public void sampleAxes() {
        if (commands.isPaused() || commands.gameplayStopped() || commands.weaponSelection()) { resetMovement();dashPointer=ultimatePointer=-1;return; }
        if (pointer >= 0) {
            float len2 = joyX * joyX + joyY * joyY;
            axisX = len2 < DEADZONE*DEADZONE ? 0 : joyX;
            axisY = len2 < DEADZONE*DEADZONE ? 0 : joyY;
        } else {
            axisX = (keys[Input.Keys.D] || keys[Input.Keys.RIGHT] ? 1 : 0) - (keys[Input.Keys.A] || keys[Input.Keys.LEFT] ? 1 : 0);
            axisY = (keys[Input.Keys.W] || keys[Input.Keys.UP] ? 1 : 0) - (keys[Input.Keys.S] || keys[Input.Keys.DOWN] ? 1 : 0);
            if (axisX != 0 && axisY != 0) { axisX *= .70710678f; axisY *= .70710678f; }
        }
    }
    public float axisX() { return axisX; }
    public float axisY() { return axisY; }
    public float knobX() { return joyX * JOY_RADIUS; }
    public float knobY() { return joyY * JOY_RADIUS; }
    public int activePointer() {return pointer;}
    public boolean joystickActive() {return pointer>=0;}
    public float centerX() {return centerX;}
    public float centerY() {return centerY;}
    public boolean inJoystickRegion(float x,float y) {
        return x>=0 && x<=hud.width()*TOUCH_WIDTH_FRACTION && y>=BOTTOM_RESERVED && y<=hud.height()-TOP_RESERVED;
    }
    @Override public boolean keyDown(int keycode) {
        // Independent latches survive movement reset. OS key-repeat cannot turn a held key into new presses.
        if(keycode==Input.Keys.SHIFT_LEFT || keycode==Input.Keys.SHIFT_RIGHT || keycode==Input.Keys.SPACE) {
            boolean held=keycode==Input.Keys.SPACE?spaceHeld:keycode==Input.Keys.SHIFT_LEFT?leftShiftHeld:rightShiftHeld;
            if(keycode==Input.Keys.SPACE)spaceHeld=true;else if(keycode==Input.Keys.SHIFT_LEFT)leftShiftHeld=true;else rightShiftHeld=true;
            if(!held && !commands.isPaused() && !commands.gameplayStopped() && !commands.weaponSelection()) {
                sampleAxes();commands.activateSkill(keycode==Input.Keys.SPACE?SkillType.ULTIMATE:SkillType.DASH,axisX,axisY);
            }
            return true;
        }
        if(commands.gameOver() || commands.victory()) {
            if(keycode==Input.Keys.ENTER || keycode==Input.Keys.R)commands.retry();
            else if(keycode==Input.Keys.ESCAPE || keycode==Input.Keys.BACK || keycode==Input.Keys.H)commands.home();
            return true;
        }
        if(commands.weaponSelection() && !commands.isPaused()) {
            if(keycode==Input.Keys.BACK || keycode==Input.Keys.ESCAPE || keycode==Input.Keys.P)commands.togglePause();
            else if(keycode==Input.Keys.H)commands.home();
            else if(keycode>=Input.Keys.NUM_1 && keycode<=Input.Keys.NUM_3)select(keycode-Input.Keys.NUM_1,commands.weaponToken());
            return true;
        }
        if(commands.isPaused() && keycode==Input.Keys.H) {commands.home();return true;}
        if (keycode == Input.Keys.P || keycode == Input.Keys.ESCAPE || keycode == Input.Keys.BACK) {
            reset();commands.togglePause(); return true;
        }
        if(commands.isPaused() || commands.gameplayStopped())return true;
        if (keycode == Input.Keys.F2) { if(commands.debugEnabled())commands.cycleAnimation(); return true; }
        if (keycode == Input.Keys.F3) { if(commands.debugEnabled())commands.cycleDirection(); return true; }
        if (keycode >= 0 && keycode < keys.length) keys[keycode] = true;
        return true;
    }
    @Override public boolean keyUp(int keycode) {
        if(keycode==Input.Keys.SHIFT_LEFT)leftShiftHeld=false;
        else if(keycode==Input.Keys.SHIFT_RIGHT)rightShiftHeld=false;
        else if(keycode==Input.Keys.SPACE)spaceHeld=false;
        if (keycode >= 0 && keycode < keys.length) keys[keycode] = false;
        return true;
    }
    @Override public boolean touchDown(int x, int y, int pointerId, int button) {
        if(pointerId<0 || pointerId>=heldPointers.length)return false;
        if(heldPointers[pointerId])return true;
        heldPointers[pointerId]=true;
        hud.unproject(x, y, touch);
        if(commands.gameOver() || commands.isPaused() || commands.gameplayStopped() || commands.weaponSelection())resetMovement();
        if(commands.gameOver() || commands.victory()) {
            if(hud.inResumeButton(touch.x,touch.y))commands.retry();
            else if(hud.inHomeButton(touch.x,touch.y))commands.home();
            return true;
        }
        if (commands.isPaused()) {
            if (hud.inResumeButton(touch.x, touch.y) || hud.inPauseButton(touch.x,touch.y)) commands.togglePause();
            else if(hud.inHomeButton(touch.x,touch.y))commands.home();
            return true;
        }
        if(commands.weaponSelection()) {
            if(hud.weaponHomeAt(touch.x,touch.y)) {commands.home();return true;}
            if(modalPointer<0 && com.badlogic.gdx.utils.TimeUtils.nanoTime()>=selectionBlockedUntil) {
                modalPointer=pointerId;modalToken=commands.weaponToken();selectedOffer=hud.weaponOfferAt(touch.x,touch.y,commands.weaponOfferCount());
            }
            return true;
        }
        if (hud.inPauseButton(touch.x, touch.y)) { reset();commands.togglePause(); return true; }
        if(commands.gameplayStopped())return true;
        SkillType skill=hud.skillAt(touch.x,touch.y);
        if(skill!=null) {
            if(skill==SkillType.DASH && dashPointer<0)dashPointer=pointerId;
            else if(skill==SkillType.ULTIMATE && ultimatePointer<0)ultimatePointer=pointerId;
            sampleAxes();commands.activateSkill(skill,axisX,axisY);return true;
        }
        if (commands.debugEnabled() && hud.inAnimationButton(touch.x, touch.y)) { commands.cycleAnimation(); return true; }
        if (commands.debugEnabled() && hud.inDirectionButton(touch.x, touch.y)) { commands.cycleDirection(); return true; }
        if (pointer < 0 && inJoystickRegion(touch.x,touch.y)) {
            pointer=pointerId;centerX=touch.x;centerY=touch.y;joyX=joyY=axisX=axisY=0;return true;
        }
        return false;
    }
    @Override public boolean touchDragged(int x, int y, int pointerId) {
        if(pointerId==dashPointer || pointerId==ultimatePointer)return true;
        if(commands.weaponSelection() || commands.isPaused() || commands.gameplayStopped()) {resetMovement();return true;}
        if (pointerId != pointer) return false;
        hud.unproject(x, y, touch); updateJoystick(touch.x, touch.y); return true;
    }
    private void updateJoystick(float x, float y) {
        joyX = (x - centerX) / JOY_RADIUS; joyY = (y - centerY) / JOY_RADIUS;
        float len2 = joyX * joyX + joyY * joyY;
        if (len2 > 1) { float inv = 1f / (float) Math.sqrt(len2); joyX *= inv; joyY *= inv; }
    }
    @Override public boolean touchUp(int x, int y, int pointerId, int button) {
        if(pointerId>=0 && pointerId<heldPointers.length)heldPointers[pointerId]=false;
        if(pointerId==dashPointer) {dashPointer=-1;return true;}
        if(pointerId==ultimatePointer) {ultimatePointer=-1;return true;}
        if(pointerId==modalPointer) {
            int chosen=selectedOffer;long token=modalToken;modalPointer=-1;selectedOffer=-1;hud.unproject(x,y,touch);
            if(commands.weaponSelection() && !commands.isPaused() && chosen>=0 && chosen==hud.weaponOfferAt(touch.x,touch.y,commands.weaponOfferCount()))select(chosen,token);
            return true;
        }
        if(commands.weaponSelection())return true;
        if (pointerId != pointer) return false;
        releaseJoystick();return true;
    }
    @Override public boolean touchCancelled(int x, int y, int pointerId, int button) {
        if(pointerId>=0 && pointerId<heldPointers.length)heldPointers[pointerId]=false;
        if(pointerId==dashPointer) {dashPointer=-1;return true;}
        if(pointerId==ultimatePointer) {ultimatePointer=-1;return true;}
        if(pointerId==modalPointer) {modalPointer=-1;selectedOffer=-1;return true;}
        if(pointerId==pointer)releaseJoystick();return true;
    }

    public void update(float delta,float axisX,float axisY,CollisionGrid grid,float dashSeconds,float dashDX,float dashDY) {
        movement.clear();movement.add(x,y);
        immunity=Math.max(0,immunity-delta);cooldown=Math.max(0,cooldown-delta);
        if(state==State.DEATH) { animationTime+=delta;return; }
        boolean action=state==State.ATTACK || state==State.HURT;
        if(action) {
            animationTime+=delta;
            if(state==State.ATTACK && pendingHit && animationTime>=config.swordHitTime) {pendingHit=false;hitReady=true;}
            if(animationTime>=duration(state)) {enter(State.IDLE);action=false;}
        }
        float length2 = axisX * axisX + axisY * axisY;
        if (length2 > 1) { float inv = 1f / (float) Math.sqrt(length2); axisX *= inv; axisY *= inv; }
        Direction oldDirection = direction;
        float normalSeconds=Math.max(0,delta-dashSeconds);
        if(length2>.0001f && normalSeconds>0)movementDirection=Direction.fromVector(axisX,axisY);
        if(!action)direction=dashSeconds>0?Direction.fromVector(dashDX,dashDY):movementDirection;
        float oldX = x, oldY = y;
        move(dashDX,dashDY,grid);
        float normalX=x,normalY=y;
        move(axisX*stats.movementSpeed()*statuses.movementMultiplier()*normalSeconds,axisY*stats.movementSpeed()*statuses.movementMultiplier()*normalSeconds,grid);
        float movedX=x-normalX,movedY=y-normalY,length=(float)Math.hypot(movedX,movedY);
        if(length>.0001f) {lastMoveX=movedX/length;lastMoveY=movedY/length;}
        boolean moved = Math.abs(x - oldX) + Math.abs(y - oldY) > .0001f;
        if (!action) {
            if (walking != moved || direction != oldDirection) animationTime = 0;
            enter(moved?State.WALK:State.IDLE);
            animationTime += delta;
        }
        walking = moved;
    }
    private void move(float dx,float dy,CollisionGrid grid) {
        int steps = Math.max(1, (int) Math.ceil(Math.max(Math.abs(dx), Math.abs(dy)) / MAX_STEP));
        dx /= steps; dy /= steps;
        for (int i = 0; i < steps; i++) {
            if (dx!=0 && grid.canOccupy(x + dx, y, HALF_WIDTH, HALF_HEIGHT)) {x += dx;movement.add(x,y);}
            if (dy!=0 && grid.canOccupy(x, y + dy, HALF_WIDTH, HALF_HEIGHT)) {y += dy;movement.add(x,y);}
        }
    }
    /** Swept feet AABB along actual movement, with a ground LOS guard against corner/wall pickup. */
    public boolean touchesMovement(float px,float py,float hw,float hh,CollisionGrid grid) {
        float ex=HALF_WIDTH+hw,ey=HALF_HEIGHT+hh;
        for(int i=0;i<movement.size;i+=2) {
            float ax=movement.items[i],ay=movement.items[i+1];
            float bx=i+2<movement.size?movement.items[i+2]:ax,by=i+2<movement.size?movement.items[i+3]:ay;
            float dx=bx-ax,dy=by-ay,lo=0,hi=1;
            if(dx==0) {if(Math.abs(ax-px)>ex)continue;}
            else {float a=(px-ex-ax)/dx,b=(px+ex-ax)/dx;lo=Math.max(lo,Math.min(a,b));hi=Math.min(hi,Math.max(a,b));}
            if(dy==0) {if(Math.abs(ay-py)>ey)continue;}
            else {float a=(py-ey-ay)/dy,b=(py+ey-ay)/dy;lo=Math.max(lo,Math.min(a,b));hi=Math.min(hi,Math.max(a,b));}
            if(lo<=hi && grid.clearLine(ax+dx*lo,ay+dy*lo,px,py,0,0,false))return true;
        }
        return false;
    }

            dashActivation=(SkillConfig.Dash)state.tuning;dashActivationLevel=state.level;

    /** Player's existing swept movement also records every dash segment for pickup contact. */
    public void advance(float dt,float axisX,float axisY,Player player,CollisionGrid grid) {
        if(dt<=0 || !player.alive())return;
        float dashSeconds=(float)Math.min(dt,dash.remaining);
        if(dashSeconds>0) {
            if(trailCount==trailX.length) {System.arraycopy(trailX,1,trailX,0,trailX.length-1);System.arraycopy(trailY,1,trailY,0,trailY.length-1);trailCount--;}
            trailX[trailCount]=player.x;trailY[trailCount++]=player.y;dashTrail=.14f;
        } else {dashTrail=Math.max(0,dashTrail-dt);if(dashTrail==0)trailCount=0;}
        player.update(dt,axisX,axisY,grid,dashSeconds,dashX*dashActivation.distance/dashActivation.duration*dashSeconds,dashY*dashActivation.distance/dashActivation.duration*dashSeconds);
        dash.cooldown=tick(dash.cooldown,dt);dash.remaining=tick(dash.remaining,dt);
        ultimate.cooldown=tick(ultimate.cooldown,dt);ultimateFade=Math.max(0,ultimateFade-dt);
        if(ultimate.remaining>0) {ultimate.remaining=tick(ultimate.remaining,dt);if(ultimate.remaining==0)ultimate.pendingImpact=true;}
    }
    /** Exactly one target scan when the cast releases. Damage hook owns kills/drops/Boss guards. */
    public void resolve(Player player,EnemySystem enemies,CollisionGrid grid,ProjectileSystem effects) {
        if(!ultimate.pendingImpact || !player.alive() || impactedId==ultimate.activationId)return;
        ultimate.pendingImpact=false;impactedId=ultimate.activationId;ultimateImpacts++;ultimateFade=.28f;
        audio.play(AudioEvent.SKILL_ULTIMATE_IMPACT);int fx=0;
        for(int i=0;i<enemies.size();i++) {
            var enemy=enemies.get(i);if(!enemy.alive())continue;
            float dx=enemy.x-ultimateX,dy=enemy.y-ultimateY;
            if(dx*dx+dy*dy>ultimateActivation.radius*ultimateActivation.radius || !grid.clearLine(ultimateX,ultimateY,enemy.x,enemy.y,.5f,.5f,true))continue;
            int before=enemy.hp();boolean killed=enemy.damage(ultimateDamage);if(enemy.hp()==before)continue;
            ultimateHits++;if(killed)ultimateKills++;
            if(enemy.boss)ultimateBossHits++;
            if(effects!=null && fx++<ultimateActivation.hitFxCap)effects.effect(ProjectileSystem.EffectType.HIT,enemy.x,enemy.y,0,0,4);
            if(!enemy.boss && enemy.alive()) {
                float distance=ultimateActivation.knockback*(enemy.definition.archetype==EnemyArchetype.TANK?ultimateActivation.tankKnockbackScale:1);
                if(enemies.knockback(enemy,dx,dy,distance))knockbacks++;
            }
        }
    }
    public void cancel() {dash.remaining=ultimate.remaining=0;ultimate.pendingImpact=false;ultimateFade=dashTrail=0;trailCount=0;}
    public void reset() {
        cancel();dash.level=ultimate.level=1;dash.tuning=config.dash;ultimate.tuning=config.ultimate;dash.reset();ultimate.reset();
        dashActivation=config.dash;ultimateActivation=config.ultimate;dashActivationLevel=ultimateActivationLevel=1;
        impactedId=0;ultimateImpacts=ultimateHits=ultimateKills=ultimateBossHits=knockbacks=0;
    }
}

    public Enemy spawnBoss(EnemyDefinition body,GameConfig.EnemyStats stats,float x,float y) {
        if(body.index>=0 || bossSize()>=config.bosses.actorCap || !bossNavigation.collision.canOccupy(x,y,body.halfWidth,body.halfHeight,false))return null;
        Enemy e=new Enemy(++nextId,body,stats,assets==null?null:assets.sprite(body.spriteId),x,y);
        e.audio=audio;e.onDeath(deathListener);enemies.add(e);return e;
    }
    public void advanceDeaths(float dt) {
        for(int i=enemies.size-1;i>=0;i--) {Enemy e=enemies.get(i);if(!e.alive()) {e.advance(dt);if(e.deathFinished())enemies.removeIndex(i);}}
    }
    public void clear() {
        skills.clear();
        for(int i=0;i<enemies.size;i++) {Enemy e=enemies.get(i);e.path.clear();e.pathIndex=0;e.onDeath(null);}
        enemies.clear();
    }
    /** Explicit isolated diagnostic arena only; Boss actors and their own pool are preserved. */
    public void diagnosticClearOrdinary(){
        if(!config.debug)throw new IllegalStateException("Diagnostic only");skills.clear();
        for(int i=enemies.size-1;i>=0;i--)if(!enemies.get(i).boss){enemies.get(i).onDeath(null);enemies.removeIndex(i);}
    }
    public void beginFrame() { searchesThisFrame=0; }
    public void update(float dt,Player player) {
        long animationStart=metrics==null?0:metrics.start();
        for(int i=enemies.size-1;i>=0;i--) {
            Enemy e=enemies.get(i);e.advance(dt);
            if(e.deathFinished())enemies.removeIndex(i);
        }
        if(metrics!=null)metrics.finish(ANIMATION,animationStart);
        if(enemies.size==0)return;
        long aiStart=metrics==null?0:metrics.start();
        long children=metrics==null?0:metrics.value(ASTAR)+metrics.value(SEPARATION)+metrics.value(COLLISION);
        long separationStart=metrics==null?0:metrics.start();
        separate();
        if(metrics!=null)metrics.finish(SEPARATION,separationStart);
        int first=rotation++%enemies.size;
        for(int n=0;n<enemies.size;n++) {
            Enemy e=enemies.get((first+n)%enemies.size);
            if(!e.alive() || e.acting() || e.aiLocked)continue;
            steer(e,player,dt);
        }
        maxSearchesObserved=Math.max(maxSearchesObserved,searchesThisFrame);
        if(aiStart!=0)metrics.add(AI,System.nanoTime()-aiStart-(metrics.value(ASTAR)+metrics.value(SEPARATION)+metrics.value(COLLISION)-children));
    }
    private void separate() {
        for(int i=0;i<enemies.size;i++) {Enemy e=enemies.get(i);e.separationX=e.separationY=0;}
        for(int i=0;i<enemies.size;i++) {
            Enemy a=enemies.get(i);if(!a.alive())continue;
            for(int j=i+1;j<enemies.size;j++) {
                if(metrics!=null && metrics.active)metrics.separationPairs++;
                Enemy b=enemies.get(j);if(!b.alive())continue;
                float dx=a.x-b.x,dy=a.y-b.y,d2=dx*dx+dy*dy;
                if(d2>=24*24)continue;
                if(d2<.001f) {dx=((a.id+b.id)&1)==0?1:-1;dy=0;d2=1;}
                float distance=(float)Math.sqrt(d2),weight=(24-distance)/24;
                dx=dx/distance*weight;dy=dy/distance*weight;
                a.separationX+=dx;a.separationY+=dy;b.separationX-=dx;b.separationY-=dy;
            }
        }
    }
    private void steer(Enemy e,Player player,float dt) {
        NavigationGrid navigation=e.boss?bossNavigation:this.navigation;
        float dx=player.x-e.x,dy=player.y-e.y,distance2=dx*dx+dy*dy;
        int target=e.boss?navigation.approach(player.x,player.y,false):navigation.cell(player.x,player.y);
        if(e.lineTimer<=0) {
            e.lineTimer=.22f+(e.id%5)*.02f;
            e.direct=navigation.collision.clearLine(e.x,e.y,player.x,player.y,e.definition.halfWidth,e.definition.halfHeight,e.definition.flying);
            if(e.direct) {e.path.clear();e.pathIndex=0;}
        }
        float tx=player.x,ty=player.y;
        if(!e.direct) {
            int changed=e.targetCell<0?100:Math.abs(target%navigation.collision.width-e.targetCell%navigation.collision.width)
                +Math.abs(target/navigation.collision.width-e.targetCell/navigation.collision.width);
            if((e.pathTimer<=0 || changed>=2) && searchesThisFrame<Math.min(2,config.pathSearchesPerFrame)) {
                searchesThisFrame++;totalSearches++;
                int start=navigation.cell(e.x,e.y);
                long searchStart=metrics==null?0:metrics.start();
                boolean found=(e.boss?bossPathfinder:pathfinder).find(start,target,e.definition.flying,e.path);
                if(metrics!=null)metrics.finish(ASTAR,searchStart);
                e.targetCell=target;e.pathIndex=0;
                // Recenter only if needed for a tight corner; periodic refresh must not pull
                // an actor backwards to the middle of its current tile.
                if(found && e.path.size>0 && !navigation.collision.clearLine(e.x,e.y,
                    navigation.x(e.path.first()),navigation.y(e.path.first()),e.definition.halfWidth,e.definition.halfHeight,e.definition.flying))e.path.insert(0,start);
                e.pathTimer=found?.8f+(e.id%5)*.08f:1.4f;
            }
            while(e.pathIndex<e.path.size) {
                int cell=e.path.get(e.pathIndex);
                tx=navigation.x(cell);ty=navigation.y(cell);
                if(Math.abs(e.x-tx)+Math.abs(e.y-ty)>2)break;
                e.pathIndex++;
            }
            if(e.pathIndex>=e.path.size) { e.locomotion(false,0,0);return; }
        }
        dx=tx-e.x;dy=ty-e.y;
        float length=(float)Math.sqrt(dx*dx+dy*dy);
        float speed=e.stats.speed*(e.boss?1:config.enemySpeedMultiplier);
        if(e.definition.charge && e.direct && distance2>45*45 && distance2<150*150 && e.chargeCooldown<=0) {
            e.chargeTime=.7f;e.chargeCooldown=3.5f;
            audio.at(com.example.monstersurvivor.audio.AudioEvent.ENEMY_LEGACY_CHARGE,e.x,e.y,e.id);
        }
        if(e.chargeTime>0 && e.direct)speed*=1.65f;
        float vx=length>.01f?dx/length*speed:0,vy=length>.01f?dy/length*speed:0;
        if(e.direct && distance2<e.definition.contactRadius()*e.definition.contactRadius())vx=vy=0;
        if(e.standOffRange>0 && e.direct && distance2<e.standOffRange*e.standOffRange) {
            if(distance2<(e.standOffRange*.65f)*(e.standOffRange*.65f)) {vx=-dx/Math.max(.01f,length)*speed;vy=-dy/Math.max(.01f,length)*speed;}
            else vx=vy=0;
        }
        vx+=e.separationX*20;vy+=e.separationY*20;
        float v2=vx*vx+vy*vy;
        if(v2>speed*speed) {float factor=speed/(float)Math.sqrt(v2);vx*=factor;vy*=factor;}
        if(!e.direct && length<speed*dt && length>.001f) {vx=dx/dt;vy=dy/dt;}
        float oldX=e.x,oldY=e.y;
        move(e,vx*dt,vy*dt);
        boolean moved=Math.abs(e.x-oldX)+Math.abs(e.y-oldY)>.005f;
        e.locomotion(moved,e.x-oldX,e.y-oldY);
        // Invalidate stalled paths with a bounded backoff; don't alternate directions every frame.
        e.stuckTime=moved?0:e.stuckTime+dt;
        if(e.stuckTime>1 && distance2>16*16) {
            e.pathTimer=0;e.direct=false;e.lineTimer=1.2f;e.path.clear();e.pathIndex=0;e.stuckTime=0;
        }
    }
    /** Instant, swept radial displacement. Ordinary actors only; leave speed/AI tuning unchanged. */
    public boolean knockback(Enemy e,float dx,float dy,float distance) {
        if(e.boss || !e.alive() || distance<=0)return false;
        float length=(float)Math.hypot(dx,dy);if(length<.001f) {dx=1;dy=0;length=1;}
        float x=e.x,y=e.y;move(e,dx/length*distance,dy/length*distance);
        if(e.x==x && e.y==y)return false;
        skills.cancel(e);
        e.path.clear();e.pathIndex=0;e.targetCell=-1;e.pathTimer=e.lineTimer=0;e.direct=false;e.stuckTime=0;return true;
    }
    private void move(Enemy e,float dx,float dy) {
        int steps=Math.max(1,(int)Math.ceil(Math.max(Math.abs(dx),Math.abs(dy))/Player.MAX_STEP));
        dx/=steps;dy/=steps;
        for(int i=0;i<steps;i++) {
            if(navigation.collision.canOccupy(e.x+dx,e.y,e.definition.halfWidth,e.definition.halfHeight,e.definition.flying))e.x+=dx;
            if(navigation.collision.canOccupy(e.x,e.y+dy,e.definition.halfWidth,e.definition.halfHeight,e.definition.flying))e.y+=dy;
        }
    }
}

package com.example.monstersurvivor.enemy.skill;

import com.example.monstersurvivor.ai.EnemySystem;
import com.example.monstersurvivor.assets.GameAssets.State;
import com.example.monstersurvivor.config.*;

/** Part of EnemySystem: bounded casts, existing steering/animation, separate hostile projectile pool. */
public final class EnemySkillSystem {
    public static final float SIDE_MARGIN=14,BOTTOM_MARGIN=44,TOP_MARGIN=50;
    public static boolean castVisible(float x,float y,float left,float bottom,float right,float top){return x>=left+SIDE_MARGIN && x<=right-SIDE_MARGIN && y>=bottom+BOTTOM_MARGIN && y<=top-TOP_MARGIN;}
    public final EnemyProjectiles projectiles=new EnemyProjectiles();
    private final EnemySystem enemies;private final CollisionGrid grid;private final GameConfig config;private final boolean hard;
    private int rotation;private float startGate;private boolean enabled=true;
    public int starts,releases,cancelled,maxCasters,maxTelegraphs;
    public EnemySkillSystem(EnemySystem enemies,GameConfig config){this.enemies=enemies;this.config=config;grid=enemies.navigation.collision;hard=config.difficulty==Difficulty.HARD;}
    public boolean enabled(){return enabled;}
    /** Existing opt-in diagnostic callers only; never a preference or normal launch switch. */
    public void diagnosticEnabled(boolean value){if(!config.debug)throw new IllegalStateException("Diagnostic only");enabled=value;if(!value)clear();}
    public int activeCasters(){int n=0;for(int i=0;i<enemies.size();i++)if(!enemies.get(i).boss && enemies.get(i).specialSkill.busy())n++;return n;}
    public int telegraphs(){int n=0;for(int i=0;i<enemies.size();i++){var e=enemies.get(i);if(!e.boss && (e.specialSkill.phase==PREPARE || e.specialSkill.phase==RELEASE))n++;}return n;}
    public void cancel(Enemy e){if(!e.boss && e.specialSkill.busy()){e.specialSkill.cancel(e);cancelled++;}}
    public void clear(){for(int i=0;i<enemies.size();i++)cancel(enemies.get(i));projectiles.clear();startGate=0;}
    public void prepare(float dt,Player player,float progress,float left,float bottom,float right,float top){
        if(!enabled || !player.alive())return;startGate=Math.max(0,startGate-dt);
        for(int i=0;i<enemies.size();i++){
            if(!player.alive()){projectiles.clear();return;}
            var e=enemies.get(i);if(e.boss)continue;var s=e.specialSkill;var d=s.definition;
            if(d==EnemySkillDefinition.NONE)continue;
            s.cooldown=Math.max(0,s.cooldown-dt);
            e.standOffRange=progress>=d.unlock && d.kind==EnemySkillDefinition.Kind.SHOT && castVisible(e.x,e.y,left,bottom,right,top)?(hard?96:86):0;
            if(!s.busy())continue;
            if(!e.alive() || e.state()==State.HURT || Math.abs(e.x-s.originX)+Math.abs(e.y-s.originY)>4){cancel(e);continue;}
            s.age+=dt;
            if(s.phase==PREPARE){
                if(d.kind==EnemySkillDefinition.Kind.SHOT){s.aimX=player.x-e.x;s.aimY=player.y-e.y;}
                if(s.age>=s.windup){s.phase=RELEASE;s.age=0;}
            }else if(s.phase==RECOVER && s.age>=s.recovery){s.phase=COOLDOWN;s.age=0;e.aiLocked=false;e.releasePose();}
        }
        int busy=activeCasters(),warnings=telegraphs();
        int cap=progress<90?(hard?4:3):(hard?10:8),warningCap=hard?4:3;
        if(startGate>0 || busy>=cap || warnings>=warningCap || enemies.size()==0)return;
        int first=rotation++%enemies.size();
        for(int n=0;n<enemies.size();n++){
            var e=enemies.get((first+n)%enemies.size());var s=e.specialSkill;var d=s.definition;
            if(e.boss || !e.alive() || e.acting() || e.aiLocked || s.busy() || d==EnemySkillDefinition.NONE || s.cooldown>0 || progress<d.unlock)continue;
            // Leave room for UI03's opaque top panels/bottom slots and the caster's visible cue.
            if(!castVisible(e.x,e.y,left,bottom,right,top))continue;
            float dx=player.x-e.x,dy=player.y-e.y,distance2=dx*dx+dy*dy;
            if(distance2>d.range*d.range || (d.kind!=EnemySkillDefinition.Kind.SLAM && distance2<48*48))continue;
            if(!grid.clearLine(e.x,e.y,player.x,player.y,2,2,true) || d.kind==EnemySkillDefinition.Kind.SHOT && !projectiles.room(d.count))continue;
            s.phase=PREPARE;s.age=0;s.windup=d.windup(hard);s.recovery=d.recovery(hard);s.originX=e.x;s.originY=e.y;
            s.targetX=d.kind==EnemySkillDefinition.Kind.SLAM?e.x:player.x;s.targetY=d.kind==EnemySkillDefinition.Kind.SLAM?e.y:player.y;s.aimX=dx;s.aimY=dy;
            e.aiLocked=true;e.pose(e.sprites!=null && e.sprites.has(State.SPECIAL,e.direction)?State.SPECIAL:State.ATTACK,dx,dy);
            s.cooldown=d.cooldown(hard);startGate=hard?.24f:.42f;starts++;maxCasters=Math.max(maxCasters,busy+1);maxTelegraphs=Math.max(maxTelegraphs,warnings+1);break;
        }
    }
    /** After player weapons/Ultimate: a killed or knocked-back caster cannot release a pending shot. */
    public void resolve(float dt,Player player){
        if(!enabled || !player.alive())return;
        for(int i=0;i<enemies.size();i++){
            if(!player.alive()){projectiles.clear();return;}
            var e=enemies.get(i);if(e.boss)continue;var s=e.specialSkill;var d=s.definition;
            if(s.phase!=RELEASE)continue;
            if(!e.alive() || e.state()==State.HURT){cancel(e);continue;}
            if(d.kind==EnemySkillDefinition.Kind.SHOT){
                float dx=player.x-e.x,dy=player.y-e.y,length=(float)Math.hypot(dx,dy);
                // Revalidate at release; target may have escaped behind cover or reached the muzzle.
                if(length<48 || length>d.range*1.2f || !grid.clearLine(e.x,e.y,player.x,player.y,2,2,true) || !projectiles.room(d.count)){cancel(e);continue;}
                dx/=length;dy/=length;
                for(int lane=0;lane<d.count;lane++){
                    double angle=Math.toRadians((lane-(d.count-1)*.5f)*d.spreadOrRadius);float c=(float)Math.cos(angle),sn=(float)Math.sin(angle);
                    float vx=dx*c-dy*sn,vy=dx*sn+dy*c;
                    if(grid.clearLine(e.x,e.y,e.x+vx*8,e.y+vy*8,2,2,true))projectiles.fire(e,d,e.x+vx*8,e.y+vy*8,vx,vy);
                }
                projectiles.effect("cast",e.x,e.y,20,.3f);
            }else{
                EnemyProjectiles.hitArea(player,grid,s.targetX,s.targetY,d.radius(),d.damage,d.status);
                projectiles.effect(d.impact,s.targetX,s.targetY,d.radius()*2,.4f);
            }
            s.phase=RECOVER;s.age=0;s.casts++;releases++;
        }
        projectiles.update(dt,player,grid);
    }
}

package com.example.monstersurvivor.enemy.skill;

/** Ordinary-enemy pool only. A shot's velocity is written on fire and never changed during flight. */
public final class EnemyProjectiles {
    public enum Team { ENEMY }
    public static final int CAPACITY=24,FX_CAPACITY=24,AREA_CAPACITY=3;
    public static final class Shot {
        public final Team team=Team.ENEMY;
        public boolean active;public Enemy owner;public int id,ownerId,damage;
        public EnemySkillDefinition definition;
        public float x,y,vx,vy,age,lifetime,radius;
        public int direction;public float rotation;
    }
    public static final class Fx {public boolean active;public String clip;public float x,y,age,duration,size;}
    public static final class Area {public final Team team=Team.ENEMY;public boolean active;public Enemy owner;public float x,y,age,tick;}
    private final Shot[] shots=new Shot[CAPACITY];private final Fx[] effects=new Fx[FX_CAPACITY];private final Area[] areas=new Area[AREA_CAPACITY];
    public int fired,hits,blocked,expired,ownerCleared,areaHits,droppedFx;
    public EnemyProjectiles(){for(int i=0;i<shots.length;i++)shots[i]=new Shot();for(int i=0;i<effects.length;i++)effects[i]=new Fx();for(int i=0;i<areas.length;i++)areas[i]=new Area();}
    public Shot shot(int index){return shots[index];}public Fx effect(int index){return effects[index];}public Area area(int index){return areas[index];}
    public int size(){int n=0;for(var s:shots)if(s.active)n++;return n;}
    public int effectCount(){int n=0;for(var f:effects)if(f.active)n++;return n;}
    public int areaCount(){int n=0;for(var a:areas)if(a.active)n++;return n;}
    public boolean room(int count){return size()+count<=CAPACITY;}
    public boolean fire(Enemy owner,EnemySkillDefinition definition,float x,float y,float dx,float dy){
        float length=(float)Math.hypot(dx,dy);if(owner==null || owner.boss || !owner.alive() || !Float.isFinite(length) || !Float.isFinite(x) || !Float.isFinite(y) || length<.001f || definition.kind!=EnemySkillDefinition.Kind.SHOT)return false;
        for(var s:shots)if(!s.active){
            s.active=true;s.owner=owner;s.ownerId=owner.id;s.definition=definition;s.damage=definition.damage;s.radius=2;s.x=x;s.y=y;
            s.vx=dx/length*definition.speed;s.vy=dy/length*definition.speed;
            var direction=com.example.monstersurvivor.assets.GameAssets.Direction.fromVector(dx,dy);s.direction=direction.ordinal();
            s.rotation=(float)Math.toDegrees(Math.atan2(dy,dx))-switch(direction){case DOWN->-90;case LEFT->180;case RIGHT->0;case UP->90;};
            s.age=0;s.lifetime=2.6f;s.id=++fired;return true;
        }
        return false;
    }
    public void effect(String clip,float x,float y,float size,float duration){
        for(var f:effects)if(!f.active){f.active=true;f.clip=clip;f.x=x;f.y=y;f.size=size;f.age=0;f.duration=duration;return;}droppedFx++;
    }
    private void poisonArea(Enemy owner,float x,float y,CollisionGrid grid){
        if(!grid.canOccupy(x,y,2,2,true))return;
        for(var a:areas)if(!a.active){a.active=true;a.owner=owner;a.x=x;a.y=y;a.age=0;a.tick=.65f;return;}
    }
    public void clear(){for(var s:shots){s.active=false;s.owner=null;}for(var f:effects)f.active=false;for(var a:areas){a.active=false;a.owner=null;}}
    public static boolean hitArea(Player player,CollisionGrid grid,float x,float y,float radius,int damage,PlayerStatuses.Type status){
        if(!player.alive())return false;
        float dx=Math.max(0,Math.abs(player.x-x)-Player.HALF_WIDTH),dy=Math.max(0,Math.abs(player.y-y)-Player.HALF_HEIGHT);
        if(dx*dx+dy*dy>radius*radius || !grid.clearLine(x,y,player.x,player.y,.5f,.5f,true))return false;
        if(!player.damage(damage))return false;
        if(player.alive())player.statuses.apply(status);return true;
    }
    public void update(float dt,Player player,CollisionGrid grid){
        if(!player.alive()){clear();return;}
        for(var f:effects)if(f.active){f.age+=dt;if(f.age>=f.duration)f.active=false;}
        for(var a:areas)if(a.active){
            if(!a.owner.alive()){a.active=false;a.owner=null;continue;}
            float step=Math.min(dt,2.3f-a.age);a.age+=step;a.tick-=step;
            if(a.tick<=.00001f){a.tick+=1;if(hitArea(player,grid,a.x,a.y,18,1,PlayerStatuses.Type.POISON))areaHits++;}
            if(a.age>=2.3f)a.active=false;
        }
        for(var s:shots)if(s.active){
            if(!s.owner.alive()){s.active=false;s.owner=null;ownerCleared++;continue;}
            float step=Math.min(dt,s.lifetime-s.age);if(step<=0){s.active=false;expired++;continue;}
            float nx=s.x+s.vx*step,ny=s.y+s.vy*step;
            float wall=CombatGeometry.wall(grid,s.x,s.y,nx,ny,s.radius);
            float hit=CombatGeometry.hit(s.x,s.y,nx,ny,player.x,player.y,Player.HALF_WIDTH+s.radius,Player.HALF_HEIGHT+s.radius);
            float impact=hit<=wall && hit<=1?hit:wall;
            if(hit<=wall && hit<=1){
                if(player.damage(s.damage)){hits++;if(player.alive())player.statuses.apply(s.definition.status);}s.active=false;
            }else if(wall<1){blocked++;s.active=false;}
            if(!s.active){
                s.x+=(nx-s.x)*impact;s.y+=(ny-s.y)*impact;
                effect(s.definition.impact,s.x,s.y,20,.28f);
                if(s.definition.poisonArea && player.alive())poisonArea(s.owner,s.x,s.y,grid);
            }else{s.x=nx;s.y=ny;}
            s.age+=step;if(s.active && s.age>=s.lifetime){s.active=false;expired++;}
        }
    }
}

    private BossInstance place(BossDefinition definition,Player player,float left,float bottom,float right,float top) {
        spawnWindows++;var nav=enemies.bossNavigation;int goal=nav.approach(player.x,player.y,false);
        if(goal>=0) {
            for(int i=0;i<64;i++) {
                float angle=random.nextFloat()*6.2831853f;
                float x=(left+right)*.5f+(float)Math.cos(angle)*((right-left)*.5f+24);
                float y=(bottom+top)*.5f+(float)Math.sin(angle)*((top-bottom)*.5f+24);
                int cell=nav.cell(x,y);
                if(valid(cell,goal,player,left,bottom,right,top))return spawn(definition,nav.x(cell),nav.y(cell));
            }
            // Deterministic near-player fallback, never the remote corner of the 200x200 world.
            int cx=(int)(player.x/nav.collision.tileSize),cy=(int)(player.y/nav.collision.tileSize);
            for(int i=0;i<256;i++) {
                int offset=fallback++%289,x=cx-8+offset%17,y=cy-8+offset/17;
                if(x<0 || y<0 || x>=nav.collision.width || y>=nav.collision.height)continue;int cell=y*nav.collision.width+x;
                if(valid(cell,goal,player,left,bottom,right,top))return spawn(definition,nav.x(cell),nav.y(cell));
            }
        }
        spawnFailures++;return null; // SPAWNING retained; the next finite retry is scheduled above.
    }
    private boolean valid(int cell,int goal,Player player,float left,float bottom,float right,float top) {
        var nav=enemies.bossNavigation;if(!nav.connected(cell,goal,false))return false;
        float x=nav.x(cell),y=nav.y(cell),dx=x-player.x,dy=y-player.y;
        float limit=Math.max(right-left,top-bottom)*.6f+48;
        if(dx*dx+dy*dy<64*64 || dx*dx+dy*dy>limit*limit)return false;

        return nav.collision.canOccupy(x,y,9,6,false);
    }

                b.areaCasts++;b.impactTime=.25f;b.areaCooldown=!b.finalBoss()?(hard?5:6):b.phase==1?(hard?6:7):(hard?4:4.8f);
                if(b.definition.kind()==BossDefinition.Kind.ENDLESS)enemies.skills.projectiles.effect("quake",b.aimX,b.aimY,b.radius*2,.4f);
            }
        }
        if(b.finalBoss() && b.phase==2 && b.summonCooldown<=0) {
            b.summonCooldown=hard?8:10;int count=0;
            for(int i=0;i<enemies.size();i++) {var e=enemies.get(i);if(e.summonOwner==b.actor.id && e.alive())count++;}

                var e=enemies.spawn(type,x,y,progress);if(e!=null) {e.summonOwner=b.actor.id;b.summoned++;break;}
            }
        }
    }

package com.example.monstersurvivor.config;

import com.example.monstersurvivor.boss.BossDefinition;
import com.example.monstersurvivor.enemy.EnemyDefinition;
import com.example.monstersurvivor.enemy.EnemyRegistry;
import static com.example.monstersurvivor.localization.GameStrings.Key.*;

/** Endless-only growth; finite stat ceilings, unchanged resident/projectile budgets. */
public final class EndlessRules {
    public static final int BOSS_INTERVAL=300,MAX_ENEMY_HP=50_000,MAX_BOSS_HP=1_000_000;
    private static final BossConfig.Candidate[] BOSSES={
        new BossConfig.Candidate("mc_moss_golem",BOSS_MOSS),
        new BossConfig.Candidate("mc_living_armor",BOSS_ARMOR),
        new BossConfig.Candidate("mc_cyclops",BOSS_CYCLOPS),
        new BossConfig.Candidate("mc_crab_knight",BOSS_CRAB),
        new BossConfig.Candidate("rock_golem",BOSS_ROCK)
    };
    private EndlessRules() { }
    /** Zero-based growth stage. Saturating conversion also handles unrealistic diagnostic inputs. */
    public static int stage(float seconds) {
        if(Float.isNaN(seconds) || seconds<=0)return 0;
        return (int)Math.min(Integer.MAX_VALUE-1,Math.floor((double)seconds/BOSS_INTERVAL));
    }
    private static int hp(int base,double factor,int cap) {
        return (int)Math.max(1,Math.min(cap,Math.round(base*factor)));
    }
    public static GameConfig.EnemyStats enemy(GameConfig.EnemyStats base,float seconds,Difficulty difficulty) {
        int stage=stage(seconds);
        if(stage==0)return base;
        int hp=hp(base.hp,Math.pow(1.15,Math.min(stage,64)),MAX_ENEMY_HP);
        int damage=(int)Math.min(8,(long)base.damage+stage/(difficulty==Difficulty.HARD?3:4));
        return new GameConfig.EnemyStats(hp,damage,base.speed);
    }
    public static int bossCandidates(){return BOSSES.length;}
    public static BossDefinition boss(int candidate,int wave) {
        var c=BOSSES[candidate];var species=EnemyRegistry.get(c.species());
        float scale=1.65f+Math.min(.30f,Math.max(0L,(long)wave-1)*.03f);
        var body=EnemyDefinition.boss("boss_endless_"+wave+"_"+species.id,species.name,species.spriteId,
            species.scale*scale,species.feetX,species.feetY,species.attackSeconds,species.hurtSeconds,species.deathSeconds);
        return new BossDefinition(BossDefinition.Kind.ENDLESS,body,c.name());
    }
    public static GameConfig.EnemyStats bossStats(Difficulty difficulty,int wave) {
        boolean hard=difficulty==Difficulty.HARD;long growth=Math.max(0L,(long)wave-1);
        int hp=hp(hard?200:120,Math.pow(1.25,Math.min(growth,64)),MAX_BOSS_HP);
        int damage=(int)Math.min(16,(hard?3:2)+growth/(hard?2:3));
        return new GameConfig.EnemyStats(hp,damage,hard?32:30);
    }
}

            }
            if(fired) {weapon.cooldownRemaining=level.cooldown;weapon.activations++;
                if(weapon.type==WeaponType.BOW)audio.play(com.example.monstersurvivor.audio.AudioEvent.WEAPON_BOW_RELEASE);
                else if(weapon.type==WeaponType.FIREBALL)audio.play(com.example.monstersurvivor.audio.AudioEvent.WEAPON_FIRE_CAST);
            }
            else weapon.retryRemaining=config.weapons.retrySeconds;
        }
    }
    private void chain(Player player,Enemy target,WeaponInstance weapon,EnemySystem enemies,CollisionGrid grid,int damage) {
        audio.play(com.example.monstersurvivor.audio.AudioEvent.WEAPON_LIGHTNING_ACTIVATE);
        boolean landed=false;
        lightningCasts++;var level=weapon.tuning();int count=0,rootCount=0;
        // Reserve both primaries before chaining. A chain cannot steal the other initial bolt.
        while(target!=null && rootCount<level.initialBolts) {
            roots[rootCount++]=target;visited[count++]=target.id;
            target=WeaponTargeting.nearest(player.x,player.y,level.range,enemies,grid,visited,count,.5f);
        }
        for(int i=0;i<rootCount;i++) {
            target=roots[i];projectiles.effect(ProjectileSystem.EffectType.LIGHTNING,player.x,player.y,target.x,target.y,0,level);
            int before=target.hp();target.damage(damage);landed|=target.hp()<before;lightningHits++;lightningInitialBolts++;
        }
        // One shared activation budget and visited set, including both roots.
        for(int i=0;i<rootCount && count<level.targets;i++) {
            float x=roots[i].x,y=roots[i].y;
            int branch=0;
            while(count<level.targets && (!weapon.evolved() || branch<EvolutionConfig.WEB_CHAIN_PER_BRANCH) && (target=WeaponTargeting.nearest(x,y,level.chainRange,enemies,grid,visited,count,.5f))!=null) {
                branch++;
                visited[count++]=target.id;projectiles.effect(ProjectileSystem.EffectType.LIGHTNING,x,y,target.x,target.y,0,level);
                x=target.x;y=target.y;int before=target.hp();target.damage(damage);landed|=target.hp()<before;lightningHits++;
            }
        }
        java.util.Arrays.fill(roots,null);
        if(landed)audio.play(com.example.monstersurvivor.audio.AudioEvent.WEAPON_LIGHTNING_IMPACT);
    }
    private boolean volley(Player player,Enemy first,WeaponInstance weapon,EnemySystem enemies,CollisionGrid grid,int damage) {
        var level=weapon.tuning();if(projectiles.available()<level.projectileCount)return false;
        int selected=0;boolean fired=false;
        for(int i=0;i<level.projectileCount;i++) {
            Enemy target=i==0?first:WeaponTargeting.nearest(player.x,player.y,level.range,enemies,grid,visited,selected,2);
            float tx,ty;
            if(target!=null) {visited[selected++]=target.id;tx=target.x;ty=target.y;}
            else {tx=first.x;ty=first.y;}
            // Distinct lanes even with one enemy or collinear targets; collision remains swept.
            float angle=(i-(level.projectileCount-1)*.5f)*level.fanDegrees*MathUtils.degreesToRadians;
            float dx=tx-player.x,dy=ty-player.y,c=MathUtils.cos(angle),s=MathUtils.sin(angle);
            fired|=projectiles.shoot(weapon.type,player.x,player.y,player.x+dx*c-dy*s,player.y+dx*s+dy*c,damage,level);
        }
        return fired;
    }

    private void orbit(float dt,Player p,EnemySystem enemies,CollisionGrid grid,WeaponInstance weapon) {
        if(!orbitStarted) {orbitStarted=true;orbitX=p.x;orbitY=p.y;java.util.Arrays.fill(orbitIds,-1);java.util.Arrays.fill(orbitNext,0);}
        orbitTime+=dt;float turn=dt*EvolutionConfig.ORBIT_SPEED;
        // Bounded angular/translation substeps; render frame rate never decides damage frequency.
        int steps=Math.min(64,Math.max(1,(int)Math.ceil(Math.max(turn/.12f,Math.hypot(p.x-orbitX,p.y-orbitY)/4))));
        boolean landed=false;int damage=weapon.tuning().damage(p.stats.damage()),frameHits=0;
        for(int n=0;n<enemies.size();n++) {
            if(frameHits>=EvolutionConfig.ORBIT_HITS_PER_STEP)break;
            Enemy e=enemies.get(n);if(!e.alive())continue;
            int slot=-1,free=-1;for(int i=0;i<orbitIds.length;i++) {if(orbitIds[i]==e.id){slot=i;break;}if(free<0 && orbitNext[i]<=orbitTime)free=i;}
            if(slot>=0 && orbitNext[slot]>orbitTime)continue;if(slot<0)slot=free;if(slot<0)continue;
            boolean hit=false;
            for(int s=1;s<=steps && !hit;s++) {
                float t=s/(float)steps,cx=orbitX+(p.x-orbitX)*t,cy=orbitY+(p.y-orbitY)*t;
                for(int b=0;b<EvolutionConfig.ORBIT_BLADES;b++) {
                    float angle=orbitAngle+turn*t+b*MathUtils.PI;
                    float bx=cx+MathUtils.cos(angle)*EvolutionConfig.ORBIT_RADIUS,by=cy+MathUtils.sin(angle)*EvolutionConfig.ORBIT_RADIUS;
                    if(Math.abs(e.x-bx)<=e.definition.halfWidth+EvolutionConfig.ORBIT_HALF_WIDTH && Math.abs(e.y-by)<=e.definition.halfHeight+EvolutionConfig.ORBIT_HALF_WIDTH
                        && grid.clearLine(cx,cy,bx,by,.5f,.5f,true) && grid.clearLine(cx,cy,e.x,e.y,.5f,.5f,true)) {hit=true;break;}
                }
            }
            if(hit) {
                orbitIds[slot]=e.id;orbitNext[slot]=orbitTime+EvolutionConfig.ORBIT_HIT_COOLDOWN;
                int before=e.hp();e.damage(damage);landed|=e.hp()<before;orbitHits++;frameHits++;
                projectiles.effect(ProjectileSystem.EffectType.HIT,e.x,e.y,0,0,5);
            }
        }
        int before=(int)(orbitAngle/MathUtils.PI);orbitAngle=(orbitAngle+turn)%(MathUtils.PI2);
        if(before!=(int)(orbitAngle/MathUtils.PI)) {weapon.activations++;audio.play(com.example.monstersurvivor.audio.AudioEvent.WEAPON_SWORD_SWING);}
        if(landed && orbitTime>=nextOrbitSound) {audio.play(com.example.monstersurvivor.audio.AudioEvent.WEAPON_SWORD_HIT);nextOrbitSound=orbitTime+.12f;}
        weapon.cooldownRemaining=Math.max(0,EvolutionConfig.ORBIT_HIT_COOLDOWN-orbitTime%EvolutionConfig.ORBIT_HIT_COOLDOWN);
        orbitX=p.x;orbitY=p.y;
    }
    public void clearTransient() {projectiles.clear();orbitStarted=false;}

    public boolean shoot(WeaponType type,float x,float y,float tx,float ty,int damage,WeaponConfig.Level level) {
        if(projectiles.size>=config.projectileCap || (type!=WeaponType.BOW && type!=WeaponType.FIREBALL))return false;

    /** All lanes share a hit ledger: an enemy can take at most one hit per fan activation. */

    }

    public void update(float dt,EnemySystem enemies,CollisionGrid grid) {
        updateZones(dt,enemies,grid);
        for(int i=effects.size-1;i>=0;i--) {Effect e=effects.get(i);e.age+=dt;if(e.age>=e.lifetime)effectPool.free(effects.removeIndex(i));}
        for(int i=projectiles.size-1;i>=0;i--) {
            Projectile p=projectiles.get(i);float step=Math.min(dt,Math.max(0,p.lifetime-p.age));
            if(p.type==WeaponType.FIREBALL && p.speed>0)step=Math.min(step,p.travelRemaining/p.speed);
            p.age+=step;p.travelRemaining=Math.max(0,p.travelRemaining-p.speed*step);

            float wall=CombatGeometry.wall(grid,p.x,p.y,nx,ny,radius);boolean remove=false;
            // Resolve earliest enemy before the first wall, even if one frame crosses multiple tiles.
            for(int pierce=0;pierce<8 && !remove; pierce++) {
                Enemy target=null;float first=wall;
                search:for(int n=0;n<enemies.size();n++) {
                    Enemy e=enemies.get(n);if(!e.alive())continue;
                    for(int j=0;j<p.hitCount;j++)if(p.hitIds[j]==e.id)continue search;
                    if(p.volley!=null && p.volley.has(e.id))continue;
                    float t=CombatGeometry.hit(p.x,p.y,nx,ny,e.x,e.y,e.definition.halfWidth+radius,e.definition.halfHeight+radius);
                    if(t<=first && (target==null || t<first)) {first=t;target=e;}
                }
                if(target==null)break;
                float hitX=p.x+(nx-p.x)*first,hitY=p.y+(ny-p.y)*first;
                if(p.type==WeaponType.FIREBALL) {explode(p,hitX,hitY,enemies,grid);remove=true;}
                else {
                    p.hitIds[p.hitCount++]=target.id;int before=target.hp();target.damage(p.damage);hits++;
                    if(p.volley!=null)p.volley.ids[p.volley.count++]=target.id;
                    if(target.hp()<before && (p.volley==null || !p.volley.impactSound)) {
                        audio.play(com.example.monstersurvivor.audio.AudioEvent.WEAPON_ARROW_HIT);if(p.volley!=null)p.volley.impactSound=true;
                    }
                    effect(EffectType.HIT,hitX,hitY,0,0,4);
                    if(--p.remainingHits<=0)remove=true;
                }
            }
            if(!remove && wall<1) {
                if(p.type==WeaponType.FIREBALL)explode(p,p.x+(nx-p.x)*wall,p.y+(ny-p.y)*wall,enemies,grid);
                remove=true;
            }
            // A fan lane can pass beside a small target: detonate at its aimed distance.
            if(!remove && p.type==WeaponType.FIREBALL && (p.travelRemaining<=.001f || p.age>=p.lifetime)) {
                explode(p,nx,ny,enemies,grid);remove=true;
            }
            p.x=nx;p.y=ny;
            if(remove || p.age>=p.lifetime || p.x<0 || p.y<0 || p.x>=grid.width*grid.tileSize || p.y>=grid.height*grid.tileSize)
                {if(p.volley!=null)p.volley.references--;projectilePool.free(projectiles.removeIndex(i));}
        }
    }
    private void explode(Projectile p,float x,float y,EnemySystem enemies,CollisionGrid grid) {
        audio.play(com.example.monstersurvivor.audio.AudioEvent.WEAPON_FIRE_EXPLOSION);
        explosions++;effect(EffectType.EXPLOSION,x,y,0,0,p.radius);
        Effect visual=effects.peek();visual.visualScale=p.visualScale;visual.glow=p.glow;
        if(p.evolved) {
            FireZone chosen=zones[0];for(var z:zones)if(!z.active){chosen=z;break;}else if(z.age>chosen.age)chosen=z;
            chosen.x=x;chosen.y=y;chosen.radius=p.radius;chosen.age=0;chosen.damage=p.burnDamage;chosen.active=true;
        }
        // One damage event per resident, with an additional evolved blast target budget.
        int damaged=0;
        for(int i=0;i<enemies.size();i++) {
            if(p.evolved && damaged>=EvolutionConfig.METEOR_TARGETS)break;
            Enemy e=enemies.get(i);if(!e.alive())continue;float dx=e.x-x,dy=e.y-y;
            if(dx*dx+dy*dy<=p.radius*p.radius && grid.clearLine(x,y,e.x,e.y,.5f,.5f,true)) {e.damage(p.damage);hits++;damaged++;}
        }
    }
    private void updateZones(float dt,EnemySystem enemies,CollisionGrid grid) {
        for(var z:zones)if(z.active){z.age+=dt;if(z.age>=EvolutionConfig.FIRE_DURATION)z.active=false;}
        burnClock+=dt;if(burnClock<EvolutionConfig.FIRE_TICK)return;
        // No catch-up burst. Session dt is capped at .1s; overlapping zones use strongest damage once.
        burnClock%=EvolutionConfig.FIRE_TICK;boolean any=false;for(var z:zones)any|=z.active;if(!any)return;burnTicks++;
        int damaged=0;for(int i=0;i<enemies.size() && damaged<EvolutionConfig.FIRE_TICK_TARGETS;i++) {
            Enemy e=enemies.get(i);if(!e.alive())continue;int damage=0;
            for(var z:zones)if(z.active) {
                float dx=e.x-z.x,dy=e.y-z.y;
                if(dx*dx+dy*dy<=z.radius*z.radius && grid.clearLine(z.x,z.y,e.x,e.y,.5f,.5f,true))damage=Math.max(damage,z.damage);
            }
            if(damage>0){e.damage(damage);burnHits++;damaged++;}
        }
    }

/** Offers and a pending item identity, owned by the session, independent of libGDX UI. */
public final class WeaponSelection {
    private final WeaponOffer[] offers=new WeaponOffer[3];
    private static final WeaponType[] TYPES=WeaponType.values();
    private final WeaponType[] candidates=new WeaponType[TYPES.length];
    private final Random random;
    private final com.example.monstersurvivor.localization.GameStrings strings;
    // UI callbacks may outlive Retry; tokens must not repeat in a different session.
    private static final java.util.concurrent.atomic.AtomicLong NEXT_TOKEN=new java.util.concurrent.atomic.AtomicLong();
    private int count;
    private long pending=-1,token;
    public WeaponSelection(long seed) {this(seed,com.example.monstersurvivor.localization.GameStrings.of(com.example.monstersurvivor.localization.GameLanguage.VI));}
    public WeaponSelection(long seed,com.example.monstersurvivor.localization.GameStrings strings) {this.strings=strings;random=new Random(seed^0x35B100L);}
    public long pendingId() {return pending;}
    public long token() {return token;}
    public int size() {return count;}
    public WeaponOffer get(int index) {if(index<0 || index>=count)throw new IndexOutOfBoundsException();return offers[index];}
    public static boolean hasReward(WeaponLoadout loadout,Player player) {
        if(!player.alive())return false;
        for(WeaponType type:TYPES)if(loadout.canGrant(type) || loadout.canEvolve(type))return true;
        return player.stats.damage()<Integer.MAX_VALUE || player.maxHp()<Integer.MAX_VALUE || player.stats.speed()<20;
    }
    public boolean open(long id,WeaponLoadout loadout,Player player) {
        if(pending!=-1 || !player.alive())return false;
        pending=id;token=NEXT_TOKEN.incrementAndGet();count=0;int n=0;
        // Every eligible owned evolution is guaranteed on this next box (at most three).
        for(WeaponType type:TYPES)if(loadout.canEvolve(type))offers[count++]=WeaponOffer.evolution(type,strings);
        for(WeaponType type:TYPES)if(loadout.canGrant(type) && (loadout.size()>1 || type!=WeaponType.SWORD))candidates[n++]=type;
        // First box offers all three new weapons; later boxes randomize eligible unique types.
        if(loadout.size()>1)for(int i=n-1;i>0;i--) {int j=random.nextInt(i+1);WeaponType t=candidates[i];candidates[i]=candidates[j];candidates[j]=t;}
        for(int i=0;i<n && count<offers.length;i++)offers[count++]=WeaponOffer.weapon(candidates[i],loadout,strings);
        for(WeaponOffer.Kind kind:STAT_KINDS) {
            if(count==offers.length)break;WeaponOffer stat=WeaponOffer.stat(kind,strings);if(stat.valid(loadout,player))offers[count++]=stat;
        }
        if(count==0) {clear();return false;}return true;
    }
    private static final WeaponOffer.Kind[] STAT_KINDS={WeaponOffer.Kind.DAMAGE,WeaponOffer.Kind.MAX_HP,WeaponOffer.Kind.SPEED};
    public void clear() {java.util.Arrays.fill(offers,null);pending=-1;count=0;token=NEXT_TOKEN.incrementAndGet();}
}

    private void loadSprites(JsonValue root) {
        for (JsonValue v = root.get("sprites").child; v != null; v = v.next) {
                SpriteSet set = new SpriteSet(v);
                if(sets.containsKey(set.id))throw new GdxRuntimeException("Duplicate sprite id: "+set.id);
                Texture texture = manager.get(v.getString("texture"), Texture.class);
                for (JsonValue a = v.get("animations").child; a != null; a = a.next) {
                    State state = State.valueOf(a.getString("state").toUpperCase(Locale.ROOT));
                    Direction dir = Direction.valueOf(a.getString("direction").toUpperCase(Locale.ROOT));
                    Array<TextureRegion> frames = new Array<>(true, a.get("frames").size, TextureRegion[]::new);
                    for (JsonValue r = a.get("frames").child; r != null; r = r.next) {
                        JsonValue rect=r.isObject()?r.get("rect"):r;
                        Texture page=r.isObject()?manager.get(r.getString("texture"),Texture.class):texture;
                        if(rect.getInt(0)<0 || rect.getInt(1)<0 || rect.getInt(2)<=0 || rect.getInt(3)<=0
                            || rect.getInt(0)+rect.getInt(2)>page.getWidth() || rect.getInt(1)+rect.getInt(3)>page.getHeight())
                            throw new GdxRuntimeException("Frame out of bounds: "+set.id);
                        frames.add(new FrameRegion(page,rect,a.getFloat("feetX",set.feetX),
                            a.getFloat("feetY",set.feetY),a.getBoolean("mirrorX",false)));
                    }
                    if (frames.isEmpty()) throw new GdxRuntimeException("Empty animation: " + set.id);
                    set.clips[state.ordinal()][dir.ordinal()] = new Clip(new Animation<>(1f / a.getFloat("fps"), frames), a.getBoolean("loop"));
                }
                for (Direction dir : Direction.values())
                    if (!set.has(State.IDLE, dir)) throw new GdxRuntimeException("Missing idle direction: " + set.id + "/" + dir);
                sets.put(set.id, set);
        }
    }

package com.example.monstersurvivor.audio;

import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.audio.Music;
import com.badlogic.gdx.audio.Sound;
import com.badlogic.gdx.utils.*;
import com.example.monstersurvivor.gameplay.GameSession;
import com.example.monstersurvivor.boss.BossSystem.EventState;
import java.util.function.LongSupplier;
import static com.example.monstersurvivor.audio.AudioEvent.*;

/** Preloaded, path-deduplicated libGDX audio. All methods run on the game thread. */
public final class GameAudioManager implements AudioSink,Disposable {
    public static final int VOICE_CAP=16;
    public interface Loader {Sound sound(String path);Music music(String path);}
    private static final class Asset {
        final String path;
        Sound sound;Music music;
        long lastPlay=Long.MIN_VALUE;
        volatile boolean completed;
        Asset(String path) {this.path=path;}
    }
    private static final class Cue {
        Asset asset;float gain,duration;int priority,cooldown;boolean loop,enemy;
        long last=Long.MIN_VALUE;boolean logged,failed;
        int plays;
    }
    private static final class Voice {Sound sound;long id,end;int priority;float gain;}
    private final Cue[] cues=new Cue[AudioEvent.values().length];
    private final ObjectMap<String,Asset> assets=new ObjectMap<>();
    private final Voice[] voices=new Voice[VOICE_CAP];
    // Bounded per-actor reaction history, replacing oldest only after the slot is reused.
    private final int[] actors=new int[128];
    private final long[] actorTimes=new long[128];
    private final LongSupplier clock;
    private AudioSettings settings=AudioSettings.DEFAULT;
    private AudioEvent currentEvent;
    private Asset current;
    private boolean blocked,musicPaused,disposed;
    private float listenerX,listenerY;
    private long enemyLast=Long.MIN_VALUE;
    public int played,dropped,failures,musicStarts;

    public GameAudioManager() {
        this(new JsonReader().parse(Gdx.files.internal("audio/manifest.json")),new Loader() {
            public Sound sound(String path) {return Gdx.audio.newSound(Gdx.files.internal(path));}
            public Music music(String path) {return Gdx.audio.newMusic(Gdx.files.internal(path));}
        },System::nanoTime);
    }
    /** Backend/clock injection lets focused tests assert actual calls and voice eviction. */
    public GameAudioManager(JsonValue manifest,Loader loader,LongSupplier clock) {
        this.clock=clock;long start=clock.getAsLong();
        for(int i=0;i<voices.length;i++)voices[i]=new Voice();
        for(JsonValue row:manifest.get("events")) {
            AudioEvent event=AudioEvent.valueOf(row.getString("event_id"));
            Cue cue=new Cue();String path=row.getString("file");boolean music=row.getString("category").equals("MUSIC");
            Asset asset=assets.get(path);
            if(asset==null) {
                asset=new Asset(path);assets.put(path,asset);
                try {
                    if(music) {asset.music=loader.music(path);final Asset loaded=asset;asset.music.setOnCompletionListener(m -> loaded.completed=true);}
                    else asset.sound=loader.sound(path);
                } catch(RuntimeException error) {failures++;log("Load failed "+path+": "+error);}
            }
            cue.asset=asset;cue.gain=row.getFloat("gain");cue.duration=row.getFloat("duration");cue.priority=row.getInt("priority");
            cue.cooldown=row.getInt("cooldown_ms");cue.loop=row.getBoolean("loop");cue.enemy=row.getString("category").equals("ENEMY");
            if(cues[event.ordinal()]!=null)throw new IllegalArgumentException("Duplicate cue "+event);
            cues[event.ordinal()]=cue;
        }
        for(AudioEvent e:AudioEvent.values())if(cues[e.ordinal()]==null && !e.optional())throw new IllegalArgumentException("Missing required audio cue "+e);
        log("Loaded unique="+assets.size+" failures="+failures+" ms="+(clock.getAsLong()-start)/1_000_000f+" voiceCap="+VOICE_CAP);
    }
    private static void log(String message) {if(Gdx.app!=null)Gdx.app.log("GameAudio",message);}
    private void failed(Cue cue,RuntimeException error) {failures++;if(!cue.failed) {cue.failed=true;log("Playback failed "+cue.asset.path+": "+error);}}
    public void listener(float x,float y) {listenerX=x;listenerY=y;}
    public void settings(AudioSettings value) {
        if(value==null || settings==value || disposed)return;settings=value;
        if(!settings.sfxEnabled || settings.sfxVolume==0)stopVoices();
        else for(Voice v:voices)if(v.sound!=null)try {v.sound.setVolume(v.id,v.gain*settings.sfxVolume);}catch(RuntimeException error){failures++;}
        applyMusic();
    }
    @Override public void at(AudioEvent event,float x,float y,int actor) {
        Cue cue=cues[event.ordinal()];
        if(cue!=null && cue.enemy) {
            float dx=x-listenerX,dy=y-listenerY;if(dx*dx+dy*dy>220*220) {dropped++;return;}
            long now=clock.getAsLong();int slot=(actor&Integer.MAX_VALUE)%actors.length;
            // Death is allowed after a reaction; reaction/charge share a one-second actor budget.
            if(event!=ENEMY_DEATH && actors[slot]==actor && now-actorTimes[slot]<1_000_000_000L) {dropped++;return;}
            if(emit(event,now) && event!=ENEMY_DEATH) {actors[slot]=actor;actorTimes[slot]=now;}
        } else play(event);
    }
    @Override public void play(AudioEvent event) {emit(event,clock.getAsLong());}
    private boolean emit(AudioEvent event,long now) {
        if(disposed)return false;Cue cue=cues[event.ordinal()];if(cue==null) {dropped++;return false;}Asset asset=cue.asset;
        if(asset.sound==null || blocked || !settings.sfxEnabled || settings.sfxVolume==0) {dropped++;return false;}
        if(cue.last!=Long.MIN_VALUE && now-cue.last<cue.cooldown*1_000_000L
            || asset.lastPlay!=Long.MIN_VALUE && now-asset.lastPlay<35_000_000L
            || cue.enemy && enemyLast!=Long.MIN_VALUE && now-enemyLast<120_000_000L) {dropped++;return false;}
        Voice slot=null,lowest=null;
        for(Voice v:voices) {
            if(v.sound!=null && now>=v.end)stop(v);
            if(v.sound==null && slot==null)slot=v;
            else if(v.sound!=null && (lowest==null || v.priority<lowest.priority || v.priority==lowest.priority && v.end<lowest.end))lowest=v;
        }
        if(slot==null) {if(lowest==null || lowest.priority>cue.priority) {dropped++;return false;}slot=lowest;stop(slot);}
        try {
            long id=asset.sound.play(cue.gain*settings.sfxVolume);
            if(id<0) {failed(cue,new IllegalStateException("backend rejected Sound.play"));return false;}
            slot.sound=asset.sound;slot.id=id;slot.end=now+(long)((cue.duration+.04f)*1_000_000_000L);slot.priority=cue.priority;slot.gain=cue.gain;
            cue.last=asset.lastPlay=now;if(cue.enemy)enemyLast=now;played++;
            cue.plays++;
            if(!cue.logged || cue.priority>=100) {cue.logged=true;log("SFX event="+event+" id="+id+" count="+cue.plays+" path="+asset.path);}
            return true;
        } catch(RuntimeException error) {failed(cue,error);return false;}
    }
    private void stop(Voice v) {if(v.sound!=null)try {v.sound.stop(v.id);}catch(RuntimeException error){failures++;}finally {v.sound=null;}}
    public void stopVoices() {for(Voice v:voices)stop(v);}
    /** User pause/modal/background never auto-resumes simulation. The screen owns that decision. */
    public void blocked(boolean value) {
        if(disposed || blocked==value)return;blocked=value;if(value)stopVoices();applyMusic();
    }
    public void musicPaused(boolean value) {
        if(disposed || musicPaused==value)return;musicPaused=value;
        applyMusic();
    }
    public static AudioEvent musicFor(GameSession session) {
        if(session.phase()==GameSession.Phase.ENDED)return null;
        if(session.victory())return BGM_VICTORY;
        if(session.phase()==GameSession.Phase.GAME_OVER)return BGM_GAME_OVER;
        if(session.bosses.finalState()==EventState.ACTIVE || session.phase()==GameSession.Phase.WINNING)return BGM_FINAL;
        if(session.bosses.miniState()==EventState.ACTIVE)return BGM_MINI;
        return session.night.active()?BGM_NIGHT:BGM_DAY;
    }
    public void music(AudioEvent event) {
        if(disposed || event==currentEvent)return;
        Asset next=event==null || cues[event.ordinal()]==null?null:cues[event.ordinal()].asset;
        if(current!=next) {
            if(current!=null && current.music!=null)try {current.music.stop();}catch(RuntimeException error){failures++;}
            current=next;if(next!=null)next.completed=false;
        }
        currentEvent=event;
        // Same-file MINI -> FINAL updates loop/gain without stop/seek/play.
        applyMusic();log("Music state="+event+" path="+(next==null?"none":next.path));
    }
    private void applyMusic() {
        if(current==null || current.music==null || currentEvent==null || disposed)return;
        Cue cue=cues[currentEvent.ordinal()];
        try {
            Music m=current.music;m.setLooping(cue.loop);m.setVolume(settings.musicVolume*cue.gain);
            if(blocked || musicPaused || !settings.musicEnabled || settings.musicVolume==0)m.pause();
            else if(!m.isPlaying() && !current.completed) {m.play();musicStarts++;}
        }catch(RuntimeException error) {failed(cue,error);}
    }
    /** Session boundary; no resource reload and no inherited cooldown/voice/music state. */
    public void reset() {
        music(null);stopVoices();enemyLast=Long.MIN_VALUE;
        java.util.Arrays.fill(actors,0);java.util.Arrays.fill(actorTimes,0);
        for(Cue c:cues)if(c!=null)c.last=Long.MIN_VALUE;
        for(Asset a:assets.values()) {a.lastPlay=Long.MIN_VALUE;a.completed=false;}
    }
    public int activeVoices() {int count=0;long now=clock.getAsLong();for(Voice v:voices)if(v.sound!=null && v.end>now)count++;return count;}
    @Override public void dispose() {
        if(disposed)return;music(null);stopVoices();disposed=true;
        for(Asset a:assets.values()) {
            if(a.sound!=null)try {a.sound.dispose();}catch(RuntimeException error){failures++;}
            if(a.music!=null)try {a.music.dispose();}catch(RuntimeException error){failures++;}
        }
        log("Disposed played="+played+" dropped="+dropped+" failures="+failures+" musicStarts="+musicStarts);
    }
